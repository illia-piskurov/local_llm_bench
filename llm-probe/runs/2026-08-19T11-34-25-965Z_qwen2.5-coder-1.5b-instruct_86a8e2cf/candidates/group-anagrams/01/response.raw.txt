export function solve(input) {
  let groups = {};
  
  for (const word of input.words) {
    const sortedWord = word.split('').sort().join('');
    
    if (!groups[sortedWord]) {
      groups[sortedWord] = [];
    }
    
    groups[sortedWord].push(word);
  }
  
  return Object.values(groups).map(group => group.sort());
}
