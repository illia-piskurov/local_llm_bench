export function solve(input) {
  const lines = input.trim().split('\n');

  if (lines.length === 0) {
    return [];
  }

  const headers = lines[0].split(',');

  if (headers.length === 0) {
    return [];
  }


  const result = [];
  for (let i = 1; i < lines.length; i++) {
    const values = lines[i].trim().split('"');
    const obj = {};
    for (let j = 0; j < headers.length; j++) {
      obj[headers[j]] = values[j];
    }
    result.push(obj);
  }

  return result;
}
