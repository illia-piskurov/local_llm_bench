```javascript
export function solve(input) {
  let result = input.template;
  for (const key in input.data) {
    if (input.data[key] && typeof input.data[key] === 'string') {
      result = result.replace(`{{${key}} }}`, input.data[key]);
    } else if (input.data[key] !== undefined && input.data[key] !== null) {
      result = result.replace(`{{${key}} }}`);
    }
  }

  return result;
}
```
