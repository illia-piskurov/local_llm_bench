export function solve(input) {
  const locales = input.locales;
  const lang = input.lang;
  const key = input.key;
  const params = input.params;

  if (!locales[lang]) {
    return key;
  }

  let result = "";

  if (params.count === 0 && 'zero' in params) {
    result += params['zero'];
  } else if (params.count === 1 && 'one' in params) {
    result += params['one'];
  } else {
    result = params[key];
  }

  return result;
}
