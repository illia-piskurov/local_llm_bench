/**
 * Process an array of records with filtering, sorting, and pagination.
 * Synchronous ES2023 implementation.
 */
function solve(input) {
    const items = input.items;
    
    // Filter: if provided, filter items based on field value and operation
    let filteredItems = [...items]; // Create a shallow copy of the array
    if (input.filter !== undefined) {
        const { field, op, value } = input.filter;
        
        for (const item of filteredItems) {
            // Case-insensitive substring match on String(item[field])
            const val = String(item[field]).toLowerCase();
            if (!op.toLowerCase().includes('contains')) continue;
            if (value.toLowerCase() === '') continue;
            
            if (field === 'name' && !val.includes(value.toLowerCase())) {
                filteredItems.push(...items); // Keep original for stable sort
            } else if (item[field] !== value) {
                filteredItems = [];
            }
        }
    }
    
    // Sort: if provided, sort items with custom stability
    let sortedItems = [...filteredItems];
    if (input.sort !== undefined) {
        const { field, dir } = input.sort;
        
        for (let i = 0; i < sortedItems.length - 1; i++) {
            for (let j = i + 1; j < sortedItems.length; j++) {
                const itemA = sortedItems[i];
                const itemB = sortedItems[j];
                
                let cmpValA, cmpValB;
                if (itemA[field] !== undefined && itemB[field] !== undefined) {
                    cmpValA = itemA[field];
                    cmpValB = itemB[field];
                } else if (itemA[field] === undefined && itemB[field] === undefined) {
                    continue;
                } else if (itemA[field] !== undefined) {
                    cmpValA = String(itemA[field]).toLowerCase();
                } else {
                    cmpValA = '';
                }
                
                const cmpValB = String(itemB[field]).toLowerCase();
                
                // Stable sort: for ties, preserve original relative order (j > i means B comes after A)
                if ((cmpValA < cmpValB && dir === 'asc') ||
                    (cmpValA > cmpValB && dir === 'desc')) {
                    sortedItems[i] = itemB;
                    sortedItems[j] = itemA;
                } else if (cmpValA <= cmpValB) { // For ties, A comes before B (stable)

                }
            }
        }
    }
    
    // Pagination: slice to get page items and calculate total pages
    const start = Math.max(0, (input.page - 1) * input.pageSize);
    const end = start + input.pageSize;
    const paginatedItems = sortedItems.slice(start, end);
    const totalPages = Math.ceil(sortedItems.length / input.pageSize);
    
    return { items: paginatedItems, total: sortedItems.length, page: input.page, totalPages: Math.max(0, totalPages) };
}

**Time Complexity:** O(n²) for the sorting step in the worst case (bubble sort), which is acceptable for small-to-medium datasets. The filter passes through items linearly if no match is found or all pass through.

**Space Complexity:** O(n) for creating copies of arrays during filtering and sorting.
