export function solve(input) {
    const numbers = [...input.numbers];
    const target = input.target;
    const n = numbers.length;

    for (let i = 0; i < n - 1; ++i) {
        for (let j = i + 1; j < n; ++j) {
            if (numbers[i] + numbers[j] === target) {
                return [i, j];
            }
        }
    }

    return [];
}
