export function solve(input) {
    const grid = input.grid;
    const start = input.start;
    const end = input.end;

    if (grid[start[0]][start[1]] !== 0 || grid[end[0]][end[1]] !== 0) {
        return -1; // Unreachable if either start or end is blocked
    }

    const rows = grid.length;
    const cols = grid[0].length;
    const directions = [[0, 1], [1, 0], [-1, 0], [0, -1]]; // Right, Down, Up, Left

    let queue = [{ x: start[0], y: start[1], steps: 0 }];
    let visited = new Set();

    while (queue.length > 0) {
        const { x, y, steps } = queue.shift();

        if (x === end[0] && y === end[1]) {
            return steps; // Reached the end
        }

        for (const [dx, dy] of directions) {
            const newX = x + dx;
            const newY = y + dy;

            if (newX >= 0 && newX < rows && newY >= 0 && newY < cols && grid[newX][newY] === 0 && !visited.has(`${newX},${newY}`)) {
                visited.add(`${newX},${newY}`);
                queue.push({ x: newX, y: newY, steps: steps + 1 });
            }
        }
    }

    return -1; // Unreachable if no path is found
}
