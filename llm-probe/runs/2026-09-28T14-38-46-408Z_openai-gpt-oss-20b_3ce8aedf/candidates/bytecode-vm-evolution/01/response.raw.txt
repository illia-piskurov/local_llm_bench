export function solve(input) {
  const program = input.program || "";
  const lines = program.split("\n");
  const labels = new Map();
  // First pass: record label positions
  lines.forEach((line, idx) => {
    const trimmed = line.trim();
    if (trimmed === "" || trimmed.startsWith("#")) return;
    const parts = trimmed.split(/\s+/);
    if (parts[0] === "LABEL" && parts.length >= 2) {
      labels.set(parts[1], idx);
    }
  });

  const stack = [];
  const output = [];
  let pc = 0;
  let steps = 0;

  function error(msg) {
    return { output, error: msg };
  }

  while (pc < lines.length && steps < 1000) {
    const raw = lines[pc];
    const lineNum = pc + 1;
    const trimmed = raw.trim();
    if (trimmed === "" || trimmed.startsWith("#")) {
      pc++;
      continue;
    }
    const parts = trimmed.split(/\s+/);
    const instr = parts[0];
    steps++;
    switch (instr) {
      case "PUSH":
        if (parts.length < 2) return error(`Unknown instruction '${instr}' at line ${lineNum}`);
        stack.push(parseInt(parts[1], 10));
        pc++;
        break;
      case "POP":
        if (stack.length < 1) return error(`Stack underflow at line ${lineNum}`);
        stack.pop();
        pc++;
        break;
      case "ADD":
        if (stack.length < 2) return error(`Stack underflow at line ${lineNum}`);
        const bAdd = stack.pop();
        const aAdd = stack.pop();
        stack.push(aAdd + bAdd);
        pc++;
        break;
      case "SUB":
        if (stack.length < 2) return error(`Stack underflow at line ${lineNum}`);
        const bSub = stack.pop();
        const aSub = stack.pop();
        stack.push(aSub - bSub);
        pc++;
        break;
      case "MUL":
        if (stack.length < 2) return error(`Stack underflow at line ${lineNum}`);
        const bMul = stack.pop();
        const aMul = stack.pop();
        stack.push(aMul * bMul);
        pc++;
        break;
      case "DIV":
        if (stack.length < 2) return error(`Stack underflow at line ${lineNum}`);
        const bDiv = stack.pop();
        const aDiv = stack.pop();
        if (bDiv === 0) return error(`Division by zero at line ${lineNum}`);
        stack.push(Math.trunc(aDiv / bDiv));
        pc++;
        break;
      case "DUP":
        if (stack.length < 1) return error(`Stack underflow at line ${lineNum}`);
        stack.push(stack[stack.length - 1]);
        pc++;
        break;
      case "SWAP":
        if (stack.length < 2) return error(`Stack underflow at line ${lineNum}`);
        const top = stack.pop();
        const second = stack.pop();
        stack.push(top);
        stack.push(second);
        pc++;
        break;
      case "PRINT":
        if (stack.length < 1) return error(`Stack underflow at line ${lineNum}`);
        output.push(String(stack[stack.length - 1]));
        pc++;
        break;
      case "LABEL":
        // no-op
        pc++;
        break;
      case "JMP":
        if (parts.length < 2) return error(`Unknown instruction '${instr}' at line ${lineNum}`);
        if (!labels.has(parts[1])) return error(`Unknown instruction '${instr}' at line ${lineNum}`);
        pc = labels.get(parts[1]);
        break;
      case "JZ":
        if (parts.length < 2) return error(`Unknown instruction '${instr}' at line ${lineNum}`);
        if (stack.length < 1) return error(`Stack underflow at line ${lineNum}`);
        const valZ = stack.pop();
        if (valZ === 0) {
          if (!labels.has(parts[1])) return error(`Unknown instruction '${instr}' at line ${lineNum}`);
          pc = labels.get(parts[1]);
        } else {
          pc++;
        }
        break;
      case "JNZ":
        if (parts.length < 2) return error(`Unknown instruction '${instr}' at line ${lineNum}`);
        if (stack.length < 1) return error(`Stack underflow at line ${lineNum}`);
        const valNZ = stack.pop();
        if (valNZ !== 0) {
          if (!labels.has(parts[1])) return error(`Unknown instruction '${instr}' at line ${lineNum}`);
          pc = labels.get(parts[1]);
        } else {
          pc++;
        }
        break;
      default:
        return error(`Unknown instruction '${instr}' at line ${lineNum}`);
    }
  }

  if (steps >= 1000) return error("Execution step limit exceeded");
  return { output };
}
