export function solve(input) {
  const matchRule = (rule, perm) => {
    if (rule === '*') return true;
    if (rule.endsWith(':*')) {
      const prefix = rule.slice(0, -2);
      return perm.startsWith(prefix + ':');
    }
    return rule === perm;
  };

  const results = [];
  for (const req of input.requests) {
    const userDef = input.users[req.user];
    if (!userDef) {
      results.push(false);
      continue;
    }

    const denies = userDef.denies || [];
    let denied = false;
    for (const d of denies) {
      if (matchRule(d, req.permission)) {
        denied = true;
        break;
      }
    }
    if (denied) {
      results.push(false);
      continue;
    }

    const allows = [];
    if (userDef.allows) {
      allows.push(...userDef.allows);
    }
    const roles = userDef.roles || [];
    for (const r of roles) {
      const roleRules = input.roles[r] || [];
      allows.push(...roleRules);
    }

    let allowed = false;
    for (const a of allows) {
      if (matchRule(a, req.permission)) {
        allowed = true;
        break;
      }
    }

    results.push(allowed);
  }
  return results;
}
