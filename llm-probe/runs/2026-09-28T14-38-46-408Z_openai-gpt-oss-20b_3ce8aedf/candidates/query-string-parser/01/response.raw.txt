export function solve(input) {
  const query = input.query ?? "";
  const stripped = query.startsWith("?") ? query.slice(1) : query;
  if (!stripped.trim()) return {};

  const pairs = stripped.split("&");
  const result = {};

  function parseValue(v) {
    if (v === "") return "";
    if (/^-?\d+(\.\d+)?$/.test(v) && isFinite(Number(v))) return Number(v);
    if (v === "true") return true;
    if (v === "false") return false;
    return v;
  }

  function setPath(obj, parts, value) {
    const len = parts.length;
    let current = obj;
    for (let i = 0; i < len - 1; i++) {
      const { key } = parts[i];
      if (!(key in current)) current[key] = {};
      current = current[key];
    }
    const last = parts[len - 1];
    if (last.isArray) {
      if (!(last.key in current)) current[last.key] = [];
      current[last.key].push(value);
    } else {
      current[last.key] = value;
    }
  }

  for (const pair of pairs) {
    if (!pair) continue;
    const idx = pair.indexOf("=");
    let rawKey, rawVal;
    if (idx === -1) {
      rawKey = pair;
      rawVal = "true";
    } else {
      rawKey = pair.slice(0, idx);
      rawVal = pair.slice(idx + 1);
    }
    const keyDecoded = decodeURIComponent(rawKey);
    const valDecoded = decodeURIComponent(rawVal);

    // parse key into parts
    const parts = [];
    let buffer = "";
    for (let i = 0; i < keyDecoded.length; i++) {
      const ch = keyDecoded[i];
      if (ch === "[") {
        parts.push({ key: buffer, isArray: false });
        buffer = "";
      } else if (ch === "]") {
        const isArray = buffer.endsWith("[]");
        const keyName = isArray ? buffer.slice(0, -2) : buffer;
        parts.push({ key: keyName, isArray });
        buffer = "";
      } else {
        buffer += ch;
      }
    }
    if (buffer) parts.push({ key: buffer, isArray: false });

    const value = parseValue(valDecoded);
    setPath(result, parts, value);
  }

  return result;
}
