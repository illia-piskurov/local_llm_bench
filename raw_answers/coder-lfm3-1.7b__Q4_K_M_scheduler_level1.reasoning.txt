def topo_sort(tasks: dict[str, list[str]]) -> list[str]:
    # Build adjacency and indegree
    indeg = {node: 0 for node in tasks}
    adj = {node: [] for node in tasks}  # outgoing edges

    for task, deps in tasks.items():
        # For each dependency dep in deps:
        for dep in deps:
            if dep not in adj:  # might be missing? but should exist
                continue
            adj[dep].append(task)   # edge from dep -> task
            indeg[task] += 1

    # Now Kahn's algorithm
    result = []
    queue = [node for node, deg in indeg.items() if deg == 0]
    from collections import deque
    q = deque(queue)

    while q:
        u = q.popleft()
        result.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    # If we processed all nodes, return result; else cycle exists.
    if len(result) != len(tasks):
        return None
    return result
```
