
```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width,  initial-scale=1.0">
    <title>Snake Game</title>
    <style>
        body {
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            margin: 0;
            background-color: #f0f0f0;
            font-family: Arial, sans-serif;
            overflow: hidden;
        }
        
        canvas {
            border: 2px solid #333;
            background-color: #fff;
            display: block;
        }
        
        .game-container {
            position: relative;
            text-align: center;
        }
        
        .score {
            font-size: 24px;
            margin-bottom: 10px;
        }
        
        .game-over {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            background-color: rgba(0, 0, 0, 0.7);
            color: white;
            padding: 20px;
            border-radius: 10px;
            display: none;
        }
        
        .game-over button {
            background-color: #4CAF50;
            border: none;
            color: white;
            padding: 10px 20px;
            text-align: center;
            text-decoration: none;
            display: inline-block;
            font-size: 16px;
            margin: 10px 2px;
            cursor: pointer;
            border-radius: 5px;
        }
    </style>
</head>
<body>
    <div class="game-container">
        <canvas id="gameCanvas" width="400" height="300"></canvas>
        <div class="score">Score: <span id="scoreValue">0</span></div>
        <div id="gameOver" class="game-over">
            Game Over!
            <button onclick="restartGame()">Play Again</button>
        </div>
    </div>

    <script>
        // Получаем элементы
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreValue = document.getElementById('scoreValue');
        const gameOverScreen = document.getElementById('gameOver');

        // Инициализация переменных игры
        let gridSize = 20;
        let snake = [];
        let food = {};
        let direction = 'right';
        let nextDirection = 'right';
        let score = 0;
        let gameSpeed = 150; // Миллисекунды между перемещениями змеи
        let lastUpdateTime = 0;

        // Инициализация игры
        function initGame() {
            // Скрываем экран game over, если он был виден
            gameOverScreen.style.display = 'none';
            
            // Очищаем игровое поле и рисуем его
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            drawGrid();
            
            // Инициализация змеи: начальная позиция и длина
            snake = [
                {x: Math.floor(canvas.width / gridSize) * gridSize, y: Math.floor(canvas.height / gridSize) * gridSize}
            ];
            
            // Начальное направление движения
            direction = 'right';
            nextDirection = 'right';
            
            // Инициализация счета
            score = 0;
            scoreValue.textContent = score;
            
            // Создаем еду в случайном месте на игровом поле
            createFood();
            
            // Обновляем игровой цикл
            requestAnimationFrame(gameLoop);
        }

        // Рисуем сетку для игры
        function drawGrid() {
            ctx.strokeStyle = '#ddd';
            for (let x = 0; x < canvas.width; x += gridSize) {
                for (let y = 0; y < canvas.height; y += gridSize) {
                    if ((x + y) % (gridSize * 2) === 0) {
                        ctx.strokeRect(x, y, gridSize - 1, gridSize - 1);
                    }
                }
            }
        }

        // Создаем еду в случайном месте на игровом поле
        function createFood() {
            let x = Math.floor(Math.random() * (canvas.width / gridSize)) * gridSize;
            let y = Math.floor(Math.random() * (canvas.height / gridSize)) * gridSize;

            // Проверяем, не попало ли еды в змею
            for (let i = 0; i < snake.length; i++) {
                if (snake[i].x === x && snake[i].y === y) {
                    return createFood(); // Пересоздаем еду если она появилась на змее
                }
            }

            food = {x, y};
        }

        // Основной игровой цикл
        function gameLoop(timestamp) {
            if (!lastUpdateTime) lastUpdateTime = timestamp;
            const delta = timestamp - lastUpdateTime;

            // Обновляем направление движения только при достаточной скорости обработки
            if (delta > 100) {
                updateDirection();
                moveSnake(delta);
                checkCollision();
                drawEverything();
                
                // Сбрасываем временную метку для следующего кадра
                lastUpdateTime = timestamp;
            }

            requestAnimationFrame(gameLoop);
        }

        // Обновление направления движения змеи
        function updateDirection() {
            if (nextDirection === 'up' && direction !== 'down') {
                direction = 'up';
            } else if (nextDirection === 'down' && direction !== 'up') {
                direction = 'down';
            } else if (nextDirection === 'left' && direction !== 'right') {
                direction = 'left';
            } else if (nextDirection === 'right' && direction !== 'left') {
                direction = 'right';
            }
        }

        // Перемещение змеи
        function moveSnake(delta) {
            const head = {...snake[0]}; // Копируем голову
            
            switch (direction) {
                case 'up':
                    head.y -= gridSize;
                    break;
                case 'down':
                    head.y += gridSize;
                    break;
                case 'left':
                    head.x -= gridSize;
                    break;
                case 'right':
                    head.x += gridSize;
                    break;
            }
            
            // Проверяем выход за границы поля
            if (head.x < 0 || head.x >= canvas.width || 
                head.y < 0 || head.y >= canvas.height) {
                gameOver();
                return;
            }

            snake.unshift(head); // Добавляем новую голову в начало
            
            // Проверяем поедание еды
            if (head.x === food.x && head.y === food.y) {
                score++;
                scoreValue.textContent = score;
                createFood(); // Создаем новую еду
            } else {
                snake.pop(); // Удаляем хвост если не поедали еду
            }
        }

        // Проверка на столкновение с собой
        function checkCollision() {
            for (let i = 1; i < snake.length; i++) {
                if (snake[i].x === snake[0].x && snake[i].y === snake[0].y) {
                    gameOver();
                    return;
                }
            }
        }

        // Отрисовка всего на поле
        function drawEverything() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Рисуем еду
            ctx.fillStyle = 'red';
            ctx.fillRect(food.x, food.y, gridSize - 1, gridSize - 1);
            
            // Рисуем змею (голова и тело)
            snake.forEach((segment, index) => {
                if (index === 0) { // Голова
                    ctx.fillStyle = 'green';
                } else { // Тело
                    ctx.fillStyle = '#4CAF50';
                }
                
                ctx.fillRect(segment.x, segment.y, gridSize - 1, gridSize - 1);
            });
        }

        // Обработка game over
        function gameOver() {
            // Останавливаем обновление игры
            lastUpdateTime = null;
            
            // Показываем экран с результатом
            gameOverScreen.style.display = 'block';
            
            // Можно добавить звук или анимацию, но в данном случае просто останавливаем игру
        }

        // Перезапуск игры
        function restartGame() {
            initGame();
        }

        // Обработчик клавиш для управления змеей
        document.addEventListener('keydown', (event) => {
            switch(event.key.toLowerCase()) {
                case 'w':
                    nextDirection = 'up';
                    break;
                case 's':
                    nextDirection = 'down';
                    break;
                case 'a':
                    nextDirection = 'left';
                    break;
                case 'd':
                    nextDirection = 'right';
                    break;
            }
        });

        // Инициализация игры при загрузке страницы
        window.onload = initGame;

    </script>
</body>
</html>
```

Этот код реализует классическую игру "Змейка" (Snake) с следующими характеристиками:

1. Управление змеей осуществляется клавишами W, A, S, D или стрелками внизу экрана
2. Змейка растет при поедании еды и теряет сегменты при столкновении со стенами или собой
3. В левом верхнем углу отображается текущий счет (длина змеи)
4. При game over появляется сообщение "Game Over!" с кнопкой для перезапуска игры
5. Игровое поле разделено на сетку 20x20 пикселей

Игра начинается автоматически при загрузке страницы и может быть перезапущена после game over по нажатию кнопки "Play Again".
