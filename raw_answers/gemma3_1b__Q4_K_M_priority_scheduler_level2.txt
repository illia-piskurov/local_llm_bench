def critical_path(tasks: dict[str, tuple[int, list[str], int]]) -> int | None:
    """
    Implements a task scheduler with dependencies, priorities, and a limited worker pool.

    Args:
        tasks: A dictionary where keys are task names and values are tuples
               (duration, deps, priority).

    Returns:
        The length of the longest dependency path by sum of durations, ignoring priority.
        If there is a cycle in the dependency graph, return None.
    """

    in_degree = {}
    graph = {}
    for task, (duration, deps, priority) in tasks.items():
        in_degree[task] = 0
        graph[task] = []

    for task, (duration, deps, priority) in tasks.items():
        for dep in deps:
            graph[task].append(dep)
            in_degree[dep] = in_degree.get(dep, 0) + 1

    queue = []
    for task, (duration, deps, priority) in tasks.items():
        if in_degree[task] == 0:
            queue.append(task)

    result = []
    while queue:
        task = queue.pop(0)
        result.append(task)

        for neighbor in graph[task]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    if len(result) != len(tasks):
        return None

    return sum(task for task in result if task not in tasks)

def makespan(tasks: dict[str, tuple[int, list[str], int]], workers: int) -> int | None:
    """
    Returns the total completion time of all tasks using the same scheduling rules as plan_order.
    Use the same deterministic tie-breaking for ready tasks:
        descending priority, ascending duration, alphabetical name.
    If there is a cycle in the dependency graph, return None.
    """
    completion_times = {}
    for task, (duration, deps, priority) in tasks.items():
        completion_times[task] = duration
    
    current_time = 0
    for task in tasks:
        if task not in completion_times:
            return None
        current_time += completion_times[task]
    
    return current_time

def has_cycle(graph: dict[str, list[str]]) -> bool:
    """
    Detects if there is a cycle in the dependency graph.
    """
    visited = set()
    recursion_stack = set()

    def dfs(node):
        visited.add(node)
        recursion_stack.add(node)

        for neighbor in graph[node]:
            if neighbor not in visited:
                if dfs(neighbor):
                    return True
            elif neighbor in recursion_stack:
                return True

        recursion_stack.remove(node)
        return False

    for node in graph:
        if node not in visited:
            if dfs(node):
                return True

    return False
