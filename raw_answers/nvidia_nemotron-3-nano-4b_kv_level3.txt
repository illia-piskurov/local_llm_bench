```python
import copy

def run(program: str) -> list[str]:
    stack = []                     # each element is a dict for the current transaction
    global_store = {}              # final values after all commits (not used directly)
    outputs = []
    watch_set = set()              # keys that are being watched
    snapshots = {}                 # name -> snapshot dict

    for line in program.splitlines():
        if not line.strip():
            continue
        tokens = line.split()
        cmd = tokens[0]

        if cmd == "SET":
            key, value = tokens[1], tokens[2]
            if stack:
                cur_dict = stack[-1]
                old_val = cur_dict.get(key, "NULL")
                if old_val != value:                     # emit WATCH notification
                    outputs.append(f"WATCH {key} {old_val} -> {value}")
                cur_dict[key] = value

        elif cmd == "GET":
            key = tokens[1]
            val = stack[-1].get(key, "NULL") if stack else "NULL"
            outputs.append(val)

        elif cmd == "DELETE":
            key = tokens[1]
            if stack:
                cur_dict = stack[-1]
                if key in cur_dict:
                    old_val = cur_dict.pop(key)
                    outputs.append(f"WATCH {key} {old_val} -> NULL")

        elif cmd == "BEGIN":
            stack.append({})

        elif cmd == "COMMIT":
            if not stack:
                outputs.append("NO TRANSACTION")
                continue
            inner = stack.pop()
            if stack:                     # merge into parent transaction
                outer = stack[-1]
                outer.update(inner)
            else:                          # outermost transaction, effects become global
                global_store.update(inner)

        elif cmd == "ROLLBACK":
            if not stack:
                outputs.append("NO TRANSACTION")
                continue
            inner = stack.pop()           # discard changes

        elif cmd == "COUNT":
            value = tokens[1]
            cur_dict = stack[-1] if stack else {}
            cnt = sum(1 for k in cur_dict.keys() if cur_dict[k] == value)
            if not cur_dict and value == "NULL":   # count global keys with NULL value
                cnt += len(global_store)          # all global keys have NULL current value
            outputs.append(str(cnt))

        elif cmd == "WATCH":
            watch_set.add(tokens[1])

        elif cmd == "SNAPSHOT":
            name = tokens[1]
            snapshot_data = {
                'global_store': copy.deepcopy(global_store),
                'stack': [dict(d) for d in stack],
                'watch_set': set(watch_set)
            }
            snapshots[name] = snapshot_data

        elif cmd == "RESTORE":
            name = tokens[1]
            if name not in snapshots:
                continue
            snapshot_data = snapshots[name]
            global_store = snapshot_data['global_store']
            stack = snapshot_data['stack']          # replace current stack depth
            watch_set = snapshot_data['watch_set']

    return outputs
```
