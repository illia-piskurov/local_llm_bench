async function pMap(items, mapper, options) {
  const opts = options === undefined || options === null
    ? {}
    : typeof options === 'number'
      ? { concurrency: options }
      : options;

  const concurrency = opts.concurrency ?? 1;
  const retries = opts.retries ?? 0;
  const backoffMs = opts.backoffMs ?? 0;
  const timeoutMs = opts.timeoutMs ?? 0;
  const signal = opts.signal;
  const settled = opts.settled ?? false;

  if (signal && signal.aborted) {
    const reason = signal.reason ?? new Error('Aborted');
    return Promise.reject(reason);
  }

  if (items.length === 0) {
    return [];
  }

  const results = new Array(items.length);
  let pending = 0;
  let nextIndex = 0;

  async function run() {
    while (pending < concurrency) {
      if (signal?.aborted) {
        const reason = signal.reason ?? new Error('Aborted');
        throw reason;
      }

      if (nextIndex >= items.length) {
        return;
      }

      const index = nextIndex++;
      pending++;

      try {
        results[index] = await runAttempt(index);
      } catch (error) {
        if (settled) {
          results[index] = { status: 'rejected', reason: error };
        } else {
          throw error;
        }
      }

      pending--;
    }
  }

  await Promise.all(Array.from({ length: concurrency }, run));

  if (settled) {
    return results.map((entry) => entry);
  }

  return results;
}

async function runAttempt(index) {
  let remaining = retries;

  while (true) {
    let result;

    try {
      result = mapper(items[index], index);
    } catch (error) {
      if (remaining > 0) {
        remaining--;
        const delay = backoffMs * Math.pow(2, retries - remaining);
        await new Promise((resolve) => setTimeout(resolve, delay));
        continue;
      }
      throw error;
    }

    if (timeoutMs > 0) {
      const timer = new Promise((resolve) => {
        setTimeout(resolve, timeoutMs);
      });
      const timerId = timer._id;

      try {
        result = await Promise.race([result, timer.then(() => {
          throw new Error('Timeout');
        })]);
      } finally {
        clearTimeout(timerId);
      }
    }

    if (signal?.aborted) {
      const reason = signal.reason ?? new Error('Aborted');
      throw reason;
    }

    return result;
  }
}
