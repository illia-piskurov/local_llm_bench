```html
<!DOCTYPE html>
<html lang="ru">
<head>
<meta charset="UTF-8">
<title>Змейка</title>
<style>
* {
    margin: 0;
    padding: 0;
    box-sizing: border-box;
}
body {
    background: #1a1a2e;
    display: flex;
    justify-content: center;
    align-items: center;
    height: 100vh;
    font-family: 'Segoe UI', Arial, sans-serif;
    flex-direction: column;
    user-select: none;
}
#score {
    color: #e0e0e0;
    font-size: 28px;
    margin-bottom: 16px;
    letter-spacing: 2px;
}
canvas {
    border: 3px solid #e94560;
    border-radius: 4px;
    background: #16213e;
    box-shadow: 0 0 30px rgba(233, 69, 96, 0.3);
}
#message {
    color: #e94560;
    font-size: 22px;
    margin-top: 16px;
    height: 30px;
    letter-spacing: 1px;
}
</style>
</head>
<body>
<div id="score">Счёт: 0</div>
<canvas id="game" width="400" height="400"></canvas>
<div id="message"></div>
<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const scoreEl = document.getElementById('score');
const messageEl = document.getElementById('message');

const GRID = 20;
const COLS = canvas.width / GRID;
const ROWS = canvas.height / GRID;
const TICK = 120;

let snake, direction, nextDirection, food, score, alive, intervalId;

function init() {
    const mid = Math.floor(COLS / 2);
    snake = [
        { x: mid, y: Math.floor(ROWS / 2) },
        { x: mid - 1, y: Math.floor(ROWS / 2) },
        { x: mid - 2, y: Math.floor(ROWS / 2) }
    ];
    direction = { x: 1, y: 0 };
    nextDirection = { x: 1, y: 0 };
    score = 0;
    alive = true;
    scoreEl.textContent = 'Счёт: 0';
    messageEl.textContent = '';
    placeFood();
    if (intervalId) clearInterval(intervalId);
    intervalId = setInterval(tick, TICK);
}

function placeFood() {
    let pos;
    do {
        pos = {
            x: Math.floor(Math.random() * COLS),
            y: Math.floor(Math.random() * ROWS)
        };
    } while (snake.some(s => s.x === pos.x && s.y === pos.y));
    food = pos;
}

function tick() {
    if (!alive) return;

    direction = { ...nextDirection };

    const head = {
        x: snake[0].x + direction.x,
        y: snake[0].y + direction.y
    };

    if (head.x < 0 || head.x >= COLS || head.y < 0 || head.y >= ROWS) {
        return gameOver();
    }

    if (snake.some(s => s.x === head.x && s.y === head.y)) {
        return gameOver();
    }

    snake.unshift(head);

    if (head.x === food.x && head.y === food.y) {
        score += 10;
        scoreEl.textContent = 'Счёт: ' + score;
        placeFood();
    } else {
        snake.pop();
    }

    draw();
}

function gameOver() {
    alive = false;
    clearInterval(intervalId);
    messageEl.textContent = 'Game Over! Нажмите R или Пробел для перезапуска.';
    draw();
}

function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Grid lines (subtle)
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    for (let i = 0; i <= COLS; i++) {
        ctx.beginPath();
        ctx.moveTo(i * GRID, 0);
        ctx.lineTo(i * GRID, canvas.height);
        ctx.stroke();
    }
    for (let i = 0; i <= ROWS; i++) {
        ctx.beginPath();
        ctx.moveTo(0, i * GRID);
        ctx.lineTo(canvas.width, i * GRID);
        ctx.stroke();
    }

    // Food
    ctx.fillStyle = '#e94560';
    ctx.shadowColor = '#e94560';
    ctx.shadowBlur = 12;
    ctx.beginPath();
    ctx.arc(food.x * GRID + GRID / 2, food.y * GRID + GRID / 2, GRID / 2 - 2, 0, Math.PI * 2);
    ctx.fill();
    ctx.shadowBlur = 0;

    // Snake
    snake.forEach((seg, i) => {
        const ratio = 1 - i / snake.length;
        const g = Math.floor(180 + 75 * ratio);
        ctx.fillStyle = i === 0 ? '#53d769' : `rgb(30, ${g}, 80)`;
        ctx.shadowColor = i === 0 ? '#53d769' : 'transparent';
        ctx.shadowBlur = i === 0 ? 8 : 0;

        const pad = i === 0 ? 1 : 2;
        const radius = 4;
        const x = seg.x * GRID + pad;
        const y = seg.y * GRID + pad;
        const w = GRID - pad * 2;
        const h = GRID - pad * 2;

        ctx.beginPath();
        ctx.moveTo(x + radius, y);
        ctx.lineTo(x + w - radius, y);
        ctx.quadraticCurveTo(x + w, y, x + w, y + radius);
        ctx.lineTo(x + w, y + h - radius);
        ctx.quadraticCurveTo(x + w, y + h, x + w - radius, y + h);
        ctx.lineTo(x + radius, y + h);
        ctx.quadraticCurveTo(x, y + h, x, y + h - radius);
        ctx.lineTo(x, y + radius);
        ctx.quadraticCurveTo(x, y, x + radius, y);
        ctx.closePath();
        ctx.fill();
    });
    ctx.shadowBlur = 0;

    // Eyes on head
    const head = snake[0];
    ctx.fillStyle = '#fff';
    const cx = head.x * GRID + GRID / 2;
    const cy = head.y * GRID + GRID / 2;
    const eyeOff = 4;
    let e1x, e1y, e2x, e2y;
    if (direction.x === 1) {
        e1x = cx + 3; e1y = cy - eyeOff;
        e2x = cx + 3; e2y = cy + eyeOff;
    } else if (direction.x === -1) {
        e1x = cx - 3; e1y = cy - eyeOff;
        e2x = cx - 3; e2y = cy + eyeOff;
    } else if (direction.y === -1) {
        e1x = cx - eyeOff; e1y = cy - 3;
        e2x = cx + eyeOff; e2y = cy - 3;
    } else {
        e1x = cx - eyeOff; e1y = cy + 3;
        e2x = cx + eyeOff; e2y = cy + 3;
    }
    ctx.beginPath(); ctx.arc(e1x, e1y, 2.5, 0, Math.PI * 2); ctx.fill();
    ctx.beginPath(); ctx.arc(e2x, e2y, 2.5, 0, Math.PI * 2); ctx.fill();
}

document.addEventListener('keydown', e => {
    if (!alive && (e.key === 'r' || e.key === 'R' || e.key === ' ' || e.key === 'Space')) {
        init();
        return;
    }

    const key = e.key;
    if (
        (key === 'ArrowUp' || key === 'w' || key === 'W' || key === 'ц' || key === 'Ц') && direction.y !== 1 ||
        (key === 'ArrowDown' || key === 's' || key === 'S' || key === 'ы' || key === 'Ы') && direction.y !== -1 ||
        (key === 'ArrowLeft' || key === 'a' || key === 'A' || key === 'ф' || key === 'Ф') && direction.x !== 1 ||
        (key === 'ArrowRight' || key === 'd' || key === 'D' || key === 'в' || key === 'В') && direction.x !== -1
    ) {
        e.preventDefault();
    }

    if (key === 'ArrowUp' || key === 'w' || key === 'W' || key === 'ц' || key === 'Ц') {
        nextDirection = { x: 0, y: -1 };
    } else if (key === 'ArrowDown' || key === 's' || key === 'S' || key === 'ы' || key === 'Ы') {
        nextDirection = { x: 0, y: 1 };
    } else if (key === 'ArrowLeft' || key === 'a' || key === 'A' || key === 'ф' || key === 'Ф') {
        nextDirection = { x: -1, y: 0 };
    } else if (key === 'ArrowRight' || key === 'd' || key === 'D' || key === 'в' || key === 'В') {
        nextDirection = { x: 1, y: 0 };
    }
});

init();
</script>
</body>
</html>
```
