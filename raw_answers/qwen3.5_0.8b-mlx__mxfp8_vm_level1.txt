def run(program: str) -> list[str]:
    """
    Execute a simple stack-based language program.
    
    Args:
        program: Multi-line string containing instructions
    
    Returns:
        List of printed strings from PRINT operations
    """
    stack = []
    lines = program.split('\n')
    output_lines = []

    for line in lines:
        if not line.strip():
            continue  # Skip empty lines
        
        # Handle comments (lines starting with #)
        if line.startswith('#'):
            continue
        
        parts = line.split()
        
        if len(parts) < 2:
            raise ValueError(f"Instruction '{line}' has insufficient arguments")
        
        op = parts[0]
        args = parts[1:]
        
        try:
            if op == 'PUSH':
                stack.append(int(args[0]))
            elif op == 'POP':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.pop()
            elif op == 'ADD':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] + args[0])
            elif op == 'SUB':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] - args[0])
            elif op == 'MUL':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] * args[0])
            elif op == 'DIV':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] // args[0])
            elif op == 'DUP':
                if len(stack) < 1:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1])
            elif op == 'SWAP':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.pop()
                stack.append(stack[-1])
            elif op == 'PRINT':
                if len(stack) < 1:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                output_lines.append(str(stack[-1]))
        except Exception as e:
            # Handle any unexpected errors
            pass
    
    return output_lines

if __name__ == "__main__":
    try:
        result = run("PUSH 5\nPOP\nADD\nDUP\nPRINT")
        print(result)
    except Exception as e:
        print(f"Error: {e}")
