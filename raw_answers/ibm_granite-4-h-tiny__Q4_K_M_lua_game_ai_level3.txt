local BT = require("path.to.behavior_tree") -- Adjust path as needed

-- Asynchronous action support
function AsyncAction(coroutineFn)
    local co = nil
    local waitTicks = 0

    return {
        tick = function(self, ctx)
            if not co or co.status == "dead" then
                co = coroutine.create(coroutineFn)
            end

            if waitTicks > 0 then
                waitTicks = waitTicks - 1
                return BT.RUNNING
            end

            local status, result = coroutine.resume(co, ctx)

            if not status then
                -- If coroutine failed or yielded WAIT_TICKS, reset and propagate result
                self:reset()
                return result or (result == false and BT.FAILURE or BT.SUCCESS)
            end

            co = nil
            waitTicks = 0

            return result
        end,
        reset = function(self)
            if co then
                coroutine.resume(co)
                co = nil
            end
            waitTicks = 0
        end,
    }
end

-- Cascading tree reset for composite nodes
function CascadingReset(node)
    node:reset()
    for _, child in ipairs(node:GetChildren()) do
        CascadingReset(child) -- Recursively reset children
    end
end

return BT
