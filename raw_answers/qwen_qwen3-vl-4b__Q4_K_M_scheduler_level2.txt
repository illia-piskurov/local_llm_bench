def critical_path(tasks: dict[str, tuple[int, list[str]]]) -> int | None:
    if not tasks:
        return 0
    
    # Validate that all tasks have dependencies
    for task in tasks:
        if not isinstance(tasks[task], tuple) or len(tasks[task]) != 2:
            return None
    
    # Extract durations and dependencies
    durations = {}
    dependencies = {}
    for task in tasks:
        durations[task] = tasks[task][0]
        dependencies[task] = tasks[task][1]
    
    # Build graph for topological sort
    graph = {}
    in_degree = {}
    for task in tasks:
        graph[task] = dependencies[task]
        in_degree[task] = len(dependencies[task])
    
    # Check for cycles using topological sort
    topo = topo_sort(graph)
    if topo is None:
        return None
    
    # Calculate critical path using dynamic programming
    # dp[task] = earliest finish time of task
    dp = {}
    for task in tasks:
        dp[task] = 0
    
    # Process tasks in topological order
    for task in topo:
        for dep in dependencies[task]:
            dp[task] = max(dp[task], dp[dep] + durations[dep])
        dp[task] += durations[task]
    
    # Return the maximum finish time
    return max(dp.values()) if dp else 0
