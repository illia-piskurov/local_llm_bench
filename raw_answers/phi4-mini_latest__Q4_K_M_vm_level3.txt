def run(program: str) -> list[str]:
    instructions = program.strip().split('\n')
    labels = {line.split()[1]: i for i, line in enumerate(instructions, 1) if line.startswith('LABEL')}
    variables = {}
    call_stack = []
    output = []
    i = 0
    while i < len(instructions):
        instruction = instructions[i].strip()
        if not instruction or instruction.startswith('#'):
            i += 1
            continue
        parts = instruction.split()
        command = parts[0]
        if command == 'PUSH':
            stack.append(int(parts[1]))
        elif command == 'POP':
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            stack.pop()
        elif command == 'ADD':
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            a, b = stack.pop(), stack.pop()
            stack.append(a + b)
        elif command == 'SUB':
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            a, b = stack.pop(), stack.pop()
            stack.append(a - b)
        elif command == 'MUL':
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            a, b = stack.pop(), stack.pop()
            stack.append(a * b)
        elif command == 'DIV':
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            a, b = stack.pop(), stack.pop()
            if b == 0:
                raise Exception(f"Error on line {i + 1}: division by zero")
            stack.append(a // b)
        elif command == 'DUP':
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            stack.append(stack[-1])
        elif command == 'SWAP':
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            stack[-1], stack[-2] = stack[-2], stack[-1]
        elif command == 'PRINT':
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            output.append(str(stack[-1]))
        elif command.startswith('LABEL'):
            labels[parts[1]] = i
        elif command.startswith('JMP'):
            i = labels[parts[1]]
        elif command.startswith('JZ'):
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            if stack.pop() == 0:
                i = labels[parts[1]]
        elif command.startswith('JNZ'):
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            if stack.pop() != 0:
                i = labels[parts[1]]
        elif command.startswith('CALL'):
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            call_stack.append(i + 1)
            i = labels[parts[1]]
        elif command == 'RET':
            if not call_stack:
                raise Exception(f"Error on line {i + 1}: call stack underflow")
            i = call_stack.pop()
        elif command.startswith('EQ'):
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            a, b = stack.pop(), stack.pop()
            stack.append(1 if a == b else 0)
        elif command.startswith('GT'):
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            a, b = stack.pop(), stack.pop()
            stack.append(1 if a > b else 0)
        elif command.startswith('LT'):
            if len(stack) < 2:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            a, b = stack.pop(), stack.pop()
            stack.append(1 if a < b else 0)
        elif command.startswith('STORE'):
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            var = parts[1]
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            variables[var] = stack.pop()
        elif command.startswith('LOAD'):
            if not stack:
                raise Exception(f"Error on line {i + 1}: stack underflow")
            var = parts[1]
            if var not in variables:
                raise Exception(f"Error on line {i + 1}: variable {var} not stored")
            stack.append(variables[var])
        else:
            raise Exception(f"Unknown command on line {i + 1}: {command}")
        i += 1
    return output
