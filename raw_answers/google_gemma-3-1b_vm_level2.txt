def run(program: str):
    lines = program.strip().split('\n')
    stack = []
    for line in lines:
        line = line.strip()
        if not line:
            continue
        if line.startswith("#"):
            continue
        try:
            num = int(line)
            if num == 0:
                print("Error: Division by zero")
                return []
            stack.append(num)
        except ValueError:
            print("Error: Invalid number")
            return []

    result = []
    while stack:
        val = stack.pop()
        if val == 0:
            print("Error: Division by zero")
            return []
        result.append(str(val))

    print(" ".join(result))
