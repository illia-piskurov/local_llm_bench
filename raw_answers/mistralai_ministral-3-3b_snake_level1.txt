```html
<!DOCTYPE html>
<html lang="ru">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Змейка</title>
    <style>
        body {
            margin: 0;
            padding: 0;
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            background-color: #222;
            font-family: Arial, sans-serif;
            overflow: hidden;
        }
        canvas {
            border: 2px solid #444;
            background-color: #111;
        }
        .score {
            position: absolute;
            top: 10px;
            color: white;
            font-size: 24px;
            text-shadow: 1px 1px 1px black;
        }
        button {
            position: absolute;
            top: 10px;
            right: 10px;
            padding: 8px 16px;
            background-color: #333;
            color: white;
            border: none;
            cursor: pointer;
            font-size: 14px;
        }
    </style>
</head>
<body>
    <div class="score">Score: <span id="score">0</span></div>
    <button id="restartBtn">Перезапустить</button>
    <canvas id="gameCanvas" width="600" height="400"></canvas>
    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const restartBtn = document.getElementById('restartBtn');

        let gameOver = false;
        let direction = { x: 0, y: 0 };
        let nextDirection = { x: 0, y: 0 };

        const snake = [{ x: 250, y: 150 }];
        const food = { x: Math.floor(Math.random() * 24) * 25, y: Math.floor(Math.random() * 16) * 25 };
        let score = 0;

        function drawSnake() {
            ctx.fillStyle = '#4CAF50';
            snake.forEach(segment => {
                ctx.fillRect(segment.x, segment.y, 25, 25);
            });
        }

        function drawFood() {
            ctx.fillStyle = '#F44336';
            ctx.beginPath();
            const centerX = food.x + 12.5;
            const centerY = food.y + 12.5;
            const radius = 10;
            ctx.arc(centerX, centerY, radius, 0, Math.PI * 2);
            ctx.fill();
        }

        function gameLoop() {
            if (gameOver) return;

            // Очистка экрана
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Обновление направления
            direction = { ...nextDirection };

            // Поворот змейки
            const head = { x: snake[snake.length - 1].x + direction.x * 25, y: snake[snake.length - 1].y + direction.y * 25 };
            snake.unshift(head);

            // Проверка столкновения со стеной или с собой
            if (
                head.x < 0 || head.x >= canvas.width ||
                head.y < 0 || head.y >= canvas.height ||
                snake.some(segment => segment.x === head.x && segment.y === head.y)
            ) {
                gameOver = true;
                return;
            }

            // Проверка поедания еды
            if (head.x === food.x && head.y === food.y) {
                score++;
                scoreElement.textContent = score;

                // Генерация новой еды в случайном месте, не на теле змейки
                let newFoodPosition;
                do {
                    newFoodPosition = {
                        x: Math.floor(Math.random() * 24) * 25,
                        y: Math.floor(Math.random() * 16) * 25
                    };
                } while (snake.some(segment => segment.x === newFoodPosition.x && segment.y === newFoodPosition.y));

                food = newFoodPosition;
            } else {
                snake.pop();
            }

            // Рисование змейки и еды
            drawSnake();
            drawFood();

            requestAnimationFrame(gameLoop);
        }

        function handleKeyDown(e) {
            switch (e.key) {
                case 'ArrowUp':
                    if (direction.y === 0) nextDirection = { x: 0, y: -1 };
                    break;
                case 'ArrowDown':
                    if (direction.y === 0) nextDirection = { x: 0, y: 1 };
                    break;
                case 'ArrowLeft':
                    if (direction.x === 0) nextDirection = { x: -1, y: 0 };
                    break;
                case 'ArrowRight':
                    if (direction.x === 0) nextDirection = { x: 1, y: 0 };
                    break;
                case 'w':
                case 'W':
                    if (direction.y === 0) nextDirection = { x: 0, y: -1 };
                    break;
                case 's':
                case 'S':
                    if (direction.y === 0) nextDirection = { x: 0, y: 1 };
                    break;
                case 'a':
                case 'A':
                    if (direction.x === 0) nextDirection = { x: -1, y: 0 };
                    break;
                case 'd':
                case 'D':
                    if (direction.x === 0) nextDirection = { x: 1, y: 0 };
                    break;
            }
        }

        function restartGame() {
            snake.length = 1;
            snake[0] = { x: 250, y: 150 };

            food = { x: Math.floor(Math.random() * 24) * 25, y: Math.floor(Math.random() * 16) * 25 };
            score = 0;
            scoreElement.textContent = '0';
            gameOver = false;

            direction = { x: 0, y: 0 };
            nextDirection = { x: 0, y: 0 };

            restartBtn.style.display = 'none';
        }

        canvas.addEventListener('keydown', handleKeyDown);
        restartBtn.addEventListener('click', restartGame);

        // Начало игры
        gameLoop();
    </script>
</body>
</html>
```
