function pMap(items, mapper, options) {
  const defaultConcurrency = 1;
  let conv = defaultConcurrency;

  if (options !== undefined) {
    // options may be a number or an object.
    if (typeof options === 'number') {
      conv = options;
    } else if (typeof options === 'object' && options !== null) {
      if (options.hasOwnProperty('concurrency')) {
        conv = options.concurrency;
      }
    }
  }

  if (!items || items.length === 0) return [];

  const results = [];
  let index = 0;

  while (index < items.length) {
    // Determine the size of this batch: at most `conv` items.
    const batchSize = Math.min(conv, items.length - index);
    // Create promises for each item in this batch
    const promises = [];
    for (let i = 0; i < batchSize; ++i) {
      const currentItem = items[index + i];
      // Pass original index: index + i

    }
    // Wait for all mapper calls in this batch to resolve
    const batchResults = await Promise.all(promises);
    results.push(...batchResults); // push in order
    index += batchSize;
  }

  return results;
}

async function pMap(items, mapper, options) {
  let concurrency = 1;

  if (options !== undefined) {
    if (typeof options === 'number') {
      concurrency = options;
    } else if (options !== null && typeof options === 'object' && 'concurrency' in options) {
      const convOpt = options.concurrency;
      if (typeof convOpt === 'number') {
        concurrency = convOpt;
      }
    }
  }

  if (!Array.isArray(items) || items.length === 0) {
    return [];
  }

  const results = new Array(items.length);
  let idx = 0;

  while (idx < items.length) {
    const batchSize = Math.min(concurrency, items.length - idx);

    const promises = [];
    for (let i = 0; i < batchSize; ++i) {
      const currentItem = items[idx + i];
      promises.push(mapper(currentItem, idx + i));
    }

    const batchResults = await Promise.all(promises);

    for (let j = 0; j < batchSize; ++j) {
      results[idx + j] = batchResults[j];
    }

    idx += batchSize;
  }

  return results;
}
