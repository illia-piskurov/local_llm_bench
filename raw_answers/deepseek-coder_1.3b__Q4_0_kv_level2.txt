        else:
            return "NULL"

    def update_watchers(self):
        for key, (new_value, old_value) in self.watchers.items():
            if new_value != old_value:
                print(f"WATCH {key} {old_value} -> {new_value}")

    def run(self, program):
        output = super().run(program)
        self.update_watchers()
        return output
