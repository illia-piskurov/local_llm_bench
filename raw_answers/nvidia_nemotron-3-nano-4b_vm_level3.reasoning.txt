def run(program: str) -> list[str]:
    lines = program.splitlines()
    stack = []
    output = []
    label_map = {}
    call_stack = []          # stores line indices (1-based)
    memory = {}              # var name -> value

    for idx, line in enumerate(lines):
        line = line.strip()
        if not line or line.startswith('#'):
            continue
        tokens = line.split()
        op = tokens[0]

        # LABEL
        if op == 'LABEL':
            name = tokens[1]
            label_map[name] = idx + 1
            continue

        # CALL / RET
        elif op == 'CALL':
            target_name = tokens[1]
            if target_name not in label_map:
                raise ValueError(f"Line {idx+1}: undefined label '{target_name}'")
            call_stack.append(idx + 2)   # return to line after CALL (i.e., idx+2)
            idx = label_map[target_name] - 1
            continue
        elif op == 'RET':
            if not call_stack:
                raise ValueError(f"Line {idx+1}: RET with empty call stack")
            return_addr = call_stack.pop()
            idx = return_addr - 1
            continue

        # Comparison ops
        elif op in ('EQ', 'GT', 'LT'):
            if len(tokens) != 3:
                raise ValueError(f"Line {idx+1}: {op} expects two values, got {len(stack)}")
            b = stack.pop()
            a = stack.pop()
            result = 1 if (op == 'EQ' and a == b) or (op == 'GT' and a > b) or (op == 'LT' and a < b) else 0
            stack.append(result)
            continue

        # Named variables
        elif op == 'STORE':
            var = tokens[1]
            if not stack:
                raise ValueError(f"Line {idx+1}: STORE on empty stack")
            value = stack.pop()
            memory[var] = value
            continue
        elif op == 'LOAD':
            var = tokens[1]
            if var not in memory:
                raise ValueError(f"Line {idx+1}: LOAD from undefined variable '{var}'")
            stack.append(memory[var])
            continue

        # Existing instructions unchanged...
        elif op == 'PUSH':
            n = int(tokens[1])
            stack.append(n)

        elif op == 'POP':
            if not stack:
                raise ValueError(f"Line {idx+1}: POP on empty stack")
            stack.pop()

        elif op == 'ADD':
            if len(stack) < 2:
                raise ValueError(f"Line {idx+1}: ADD requires two values, got {len(stack)}")
            b = stack.pop()
            a = stack.pop()
            stack.append(a + b)

        elif op == 'SUB':
            if len(stack) < 2:
                raise ValueError(f"Line {idx+1}: SUB requires two values, got {len(stack)}")
            b = stack.pop()
            a = stack.pop()
            stack.append(a - b)

        elif op == 'MUL':
            if len(stack) < 2:
                raise ValueError(f"Line {idx+1}: MUL requires two values, got {len(stack)}")
            b = stack.pop()
            a = stack.pop()
            stack.append(a * b)

        elif op == 'DIV':
            if len(stack) < 2:
                raise ValueError(f"Line {idx+1}: DIV requires two values, got {len(stack)}")
            b = stack.pop()
            a = stack.pop()
            if b == 0:
                raise ValueError(f"Line {idx+1}: division by zero")
            stack.append(a // b)

        elif op == 'DUP':
            if not stack:
                raise ValueError(f"Line {idx+1}: DUP on empty stack")
            stack.append(stack[-1])

        elif op == 'SWAP':
            if len(stack) < 2:
                raise ValueError(f"Line {idx+1}: SWAP requires two values, got {len(stack)}")
            a = stack.pop()
            b = stack.pop()
            stack.append(a)
            stack.append(b)

        elif op == 'PRINT':
            if not stack:
                raise ValueError(f"Line {idx+1}: PRINT on empty stack")
            output.append(str(stack[-1]))

        else:
            raise ValueError(f"Line {idx+1}: unknown instruction '{op}'")

    return output
