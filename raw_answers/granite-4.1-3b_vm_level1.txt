def run(program: str) -> list[str]:
    stack = []
    output = []
    lines = [line for line in program.splitlines() if not line.strip().startswith('#') and line.strip()]
    
    for i, line in enumerate(lines, start=1):
        parts = line.split()
        command = parts[0]
        
        if command == "PUSH":
            stack.append(int(parts[1]))
        elif command == "POP":
            if not stack:
                raise ValueError(f"Line {i}: Stack is empty for POP operation.")
            stack.pop()
        elif command == "ADD":
            if len(stack) < 2:
                raise ValueError(f"Line {i}: Not enough values on the stack for ADD operation.")
            a = stack.pop()
            b = stack.pop()
            stack.append(a + b)
        elif command == "SUB":
            if len(stack) < 2:
                raise ValueError(f"Line {i}: Not enough values on the stack for SUB operation.")
            b = stack.pop()
            a = stack.pop()
            stack.append(a - b)
        elif command == "MUL":
            if len(stack) < 2:
                raise ValueError(f"Line {i}: Not enough values on the stack for MUL operation.")
            a = stack.pop()
            b = stack.pop()
            stack.append(a * b)
        elif command == "DIV":
            if len(stack) < 2:
                raise ValueError(f"Line {i}: Not enough values on the stack for DIV operation.")
            b = stack.pop()
            a = stack.pop()
            if b == 0:
                raise ValueError(f"Line {i}: Division by zero.")
            stack.append(a // b)
        elif command == "DUP":
            if not stack:
                raise ValueError(f"Line {i}: Stack is empty for DUP operation.")
            stack.append(stack[-1])
        elif command == "SWAP":
            if len(stack) < 2:
                raise ValueError(f"Line {i}: Not enough values on the stack for SWAP operation.")
            stack[-1], stack[-2] = stack[-2], stack[-1]
        elif command == "PRINT":
            if not stack:
                raise ValueError(f"Line {i}: Nothing to print (stack is empty).")
            output.append(str(stack[-1]))
    
    return output
