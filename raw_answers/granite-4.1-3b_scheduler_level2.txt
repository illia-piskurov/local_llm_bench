def critical_path( tasks: dict[str, tuple[int, list[str]]]) -> int | None:
    from collections import defaultdict

    # Получаем топологическую сортировку задач
    order = topo_sort(tasks)
    if not order:
        return None

    # Инициализация временных массивов для расчета длительности пути
    duration_at_task = {task: 0 for task in tasks}
    max_duration_at_task = {task: 0 for task in tasks}

    # Расчёт длительности каждой задачи по топологическому порядку
    for task in order:
        duration, dependencies = tasks[task]
        if all(duration_at_task[dep] != 0 for dep in dependencies):
            duration_at_task[task] = sum(duration_at_task[dep] + durations[dep] 
                                         for dep in dependencies)
        else:
            duration_at_task[task] = duration

        # Обновление максимальной длительности до текущей задачи
        max_duration_at_task[task] = max(max_duration_at_task[task],
                                         duration_at_task[task])

    return max(max_duration_at_task.values()) if max_duration_at_task else None
