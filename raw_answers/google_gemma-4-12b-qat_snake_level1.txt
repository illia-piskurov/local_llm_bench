```html
<!DOCTYPE html>
<html lang="ru">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Snake Game</title>
    <style>
        body {
            display: flex;
            flex-direction: column;
            align-items: center;
            justify-content: center;
            height: 100vh;
            margin: 0;
            background-color: #2c3e50;
            color: white;
            font-family: 'Arial', sans-serif;
            overflow: hidden;
        }
        h1 {
            margin-bottom: 10px;
        }
        #game-container {
            position: relative;
            border: 5px solid #ecf0f1;
            box-shadow: 0 0 20px rgba(0,0,0,0.5);
        }
        canvas {
            display: block;
            background-color: #000;
        }
        #score {
            font-size: 24px;
            margin-bottom: 10px;
        }
        #game-over {
            position: absolute;
            top: 50%;
            left: 50%;
            transform: translate(-50%, -50%);
            text-align: center;
            display: none;
            background: rgba(0, 0, 0, 0.8);
            padding: 20px;
            border-radius: 10px;
        }
        .controls-hint {
            margin-top: 15px;
            font-size: 14px;
            color: #bdc3c7;
        }
    </style>
</head>
<body>

    <h1>Змейка</h1>
    <div id="score">Счёт: 0</div>
    
    <div id="game-container">
        <canvas id="gameCanvas" width="400" height="400"></canvas>
        <div id="game-over">
            <h2>ИГРА ОКОНЧЕНА</h2>
            <p>Нажмите ПРОБЕЛ или R для перезапуска</p>
        </div>
    </div>

    <div class="controls-hint">Управление: Стрелки или WASD</div>

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const gameOverScreen = document.getElementById('game-over');

        const gridSize = 20;
        const tileCount = canvas.width / gridSize;

        let snake, food, dx, dy, score, gameRunning, gameSpeed;

        function initGame() {
            // Начальные координаты змейки
            snake = [
                {x: 10, y: 10},
                {x: 9, y: 10},
                {x: 8, y: 10}
            ];
            
            dx = 1; // Движение вправо
            dy = 0;
            score = 0;
            gameRunning = true;
            gameSpeed = 100; // Мс между кадрами

            scoreElement.innerText = `Счёт: ${score}`;
            gameOverScreen.style.display = 'none';
            
            createFood();
            
            if (window.requestAnimationFrame) {
                requestAnimationFrame(gameLoop);
            }
        }

        function createFood() {
            food = {
                x: Math.floor(Math.random() * tileCount),
                y: Math.floor(Math.random() * tileCount)
            };
            // Проверка, чтобы еда не появилась внутри змейки
            snake.forEach(part => {
                if (part.x === food.x && part.y === food.y) {
                    createFood();
                }
            });
        }

        function gameLoop() {
            if (!gameRunning) return;

            setTimeout(() => {
                clearCanvas();
                drawFood();
                advanceSnake();
                drawSnake();
                requestAnimationFrame(gameLoop);
            }, gameSpeed);
        }

        function clearCanvas() {
            ctx.fillStyle = 'black';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
        }

        function drawSnake() {
            ctx.fillStyle = '#2ecc71';
            snake.forEach((part, index) => {
                // Голова чуть светлее
                if (index === 0) ctx.fillStyle = '#27ae60';
                else ctx.fillStyle = '#2ecc71';
                
                ctx.fillRect(part.x * gridSize, part.y * gridSize, gridSize - 2, gridSize - 2);
            });
        }

        function advanceSnake() {
            const head = { x: snake[0].x + dx, y: snake[0].y + dy };

            // Столкновение со стенами
            if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {
                return gameOver();
            }

            // Столкновение с самим собой
            for (let i = 0; i < snake.length; i++) {
                if (snake[i].x === head.x && snake[i].y === head.y) {
                    return gameOver();
                }
            }

            snake.unshift(head);

            // Поедание еды
            if (head.x === food.x && head.y === food.y) {
                score += 10;
                scoreElement.innerText = `Счёт: ${score}`;
                createFood();
                // Увеличение скорости каждые 50 очков
                if (score % 50 === 0 && gameSpeed > 50) gameSpeed -= 5;
            } else {
                snake.pop();
            }
        }

        function drawFood() {
            ctx.fillStyle = '#e74c3c';
            ctx.fillRect(food.x * gridSize, food.y * gridSize, gridSize - 2, gridSize - 2);
        }

        function gameOver() {
            gameRunning = false;
            gameOverScreen.style.display = 'block';
        }

        function handleInput(e) {
            const key = e.key.toLowerCase();
            
            // Предотвращаем разворот на 180 градусов
            if ((key === 'arrowup' || key === 'w') && dy === 0) {
                dx = 0;
                dy = -1;
            }
            if ((key === 'arrowdown' || key === 's') && dy === 0) {
                dx = 0;
                dy = 1;
            }
            if ((key === 'arrowleft' || key === 'a') && dx === 0) {
                dx = -1;
                dy = 0;
            }
            if ((key === 'arrowright' || key === 'd') && dx === 0) {
                dx = 1;
                dy = 0;
            }

            // Перезапуск
            if (key === ' ' || key === 'r') {
                if (!gameRunning) {
                    initGame();
                }
            }
        }

        window.addEventListener('keydown', handleInput);

        // Запуск игры
        initGame();
    </script>
</body>
</html>
```
