#include <stdint.h>
#include <string.h>

#define MAGIC 0xAA
#define MAX_PACKET_SIZE 256 // Assuming a maximum packet size for the ring buffer

// RingBuffer structure and functions from Level 1 here...

int ring_buffer[MAX_PACKET_SIZE];
int head = 0;
int tail = 0;
int count = 0;

void feed_bytes(const uint8_t *data, int len) {
    for (int i = 0; i < len; i++) {
        int next_head = (head + 1) % MAX_PACKET_SIZE;
        if (next_head == tail) {
            // Buffer is full, discard incoming bytes
            return;
        }
        ring_buffer[head] = data[i];
        head = next_head;
        if (count < MAX_PACKET_SIZE) {
            count++;
        }
    }
}

int get_next_packet(uint8_t *out_payload, int *out_type) {
    int i = 0;
    int packet_len = 0;
    int payload_len = 0;
    int checksum = 0;
    int found_magic = 0;
    int found_type = 0;
    int found_len = 0;

    while (i < count) {
        if (ring_buffer[i] == MAGIC) {
            if (found_magic) {
                return -1; // Corrupted packet: multiple magic bytes
            }
            found_magic = 1;
            i += 1; // Skip magic byte

            if (i + 1 >= count) {
                return -1; // Not enough data for a full packet
            }
            *out_type = ring_buffer[i + 1];
            found_type = 1;
            i += 2; // Skip type and len

            if (i + 1 >= count) {
                return -1; // Not enough data for len
            }
            payload_len = ring_buffer[i + 1];
            found_len = 1;
            i += 2; // Skip len

            if (payload_len == 0) {
                checksum = 0;
            } else if (i + payload_len + 1 >= count) {
                return -1; // Not enough data for payload and checksum
            }
            checksum = 0;
            for (int j = 0; j < payload_len; j++) {
                checksum ^= ring_buffer[i + 2 + j];
            }
            if (checksum != ring_buffer[i + payload_len + 1]) {
                return -1; // Corrupted packet: checksum mismatch
            }
            payload_len += 1; // Include checksum byte in payload length
            i += payload_len + 1; // Skip payload and checksum
        } else {
            i += 1; // Skip noise byte
        }
    }

    if (!found_magic || !found_type || !found_len) {
        return -1; // No valid packet found
    }

    if (payload_len > *out_type) {
        return -1; // Payload length exceeds type identifier
    }

    memcpy(out_payload, &ring_buffer[head], payload_len);
    *out_type = *out_type;
    return payload_len;
}
