def run(program: str) -> list[str]:
    """
    Executes the given program string.
    Returns a list of output lines (each line is a string).
    """

    stack = []
    outputs = []
    labels = {}   # name -> 1-based line number where label is defined

    lines = program.splitlines()
    i = 0
    n = len(lines)

    while i < n:
        raw_line = lines[i].strip()

        # Skip empty lines and comments
        if not raw_line or raw_line.lstrip().startswith('#'):
            i += 1
            continue

        parts = raw_line.split()
        op = parts[0].upper() if parts else ''
        line_no = i + 1   # 1‑based line number for error messages

        # ------------------------------------------------------------------
        # LABEL: define a label (no effect on execution)
        # ------------------------------------------------------------------
        if op == 'LABEL':
            if len(parts) < 2:
                raise ValueError(f"line {line_no}: missing argument for LABEL")
            label_name = parts[1].strip().upper()
            labels[label_name] = line_no
            i += 1   # move past the definition line

        # ------------------------------------------------------------------
        # JMP: unconditional jump to a labelled line
        # ------------------------------------------------------------------
        elif op == 'JMP':
            if len(parts) < 2:
                raise ValueError(f"line {line_no}: missing argument for JMP")
            target_label = parts[1].strip().upper()
            if target_label in labels:
                # jump to the label's line and continue without advancing this iteration
                i = labels[target_label]
                continue
            else:
                raise ValueError(f"line {line_no}: unknown jump target '{target_label}'")

        # ------------------------------------------------------------------
        # JZ: pop, jump if top of stack == 0
        # ------------------------------------------------------------------
        elif op == 'JZ':
            if not stack:
                raise ValueError(f"line {line_no}: JZ needs at least one item on the stack (pop value)")
            val = stack.pop()
            target_label = parts[1].strip().upper() if len(parts) >= 2 else None
            jumped = False
            if target_label and target_label in labels:
                i = labels[target_label]
                jumped = True   # jump forward instead of falling through

            if val == 0 and not jumped:
                # No valid target label → fall through to next line (i += 1 is done below)
                pass
            elif val != 0 and jumped:
                # We already set i, so just fall through; no extra increment needed here
                pass
            else:
                # Normal flow – move forward one line
                i += 1

        # ------------------------------------------------------------------
        # JNZ: pop, jump if top of stack != 0
        # ------------------------------------------------------------------
        elif op == 'JNZ':
            if not stack:
                raise ValueError(f"line {line_no}: JNZ needs at least one item on the stack (pop value)")
            val = stack.pop()
            target_label = parts[1].strip().upper() if len(parts) >= 2 else None
            jumped = False
            if target_label and target_label in labels:
                i = labels[target_label]
                jumped = True

            if val != 0 and jumped:
                # Jump forward without extra increment (jumped is set)
                pass
            elif val == 0 or not jumped:
                i += 1   # normal execution – move to next line

        # ------------------------------------------------------------------
        # PUSH
        # ------------------------------------------------------------------
        elif op == 'PUSH':
            if len(parts) < 2:
                raise ValueError(f"line {line_no}: missing argument for PUSH (need a number)")
            try:
                n = int(parts[1])
            except ValueError:
                raise ValueError(f"line {line_no}: invalid number '{parts[1]}' for PUSH")
            stack.append(n)

        # ------------------------------------------------------------------
        # POP
        # ------------------------------------------------------------------
        elif op == 'POP':
            if not stack:
                raise RuntimeError(f"line {line_no}: POP requires at least one item on the stack")
            stack.pop()

        # ------------------------------------------------------------------
        # ADD, SUB, MUL, DIV, DUP, SWAP, CALL (if we add later)
        # ------------------------------------------------------------------
        elif op == 'ADD':
            if len(stack) < 2:
                raise ValueError(f"line {line_no}: ADD needs two operands")
            b = stack.pop()
            a = stack.pop()
            stack.append(a + b)

        elif op == 'SUB':
            if len(stack) < 2:
                raise ValueError(f"line {line_no}: SUB needs two operands")
            b = stack.pop()
            a = stack.pop()
            stack.append(a - b)

        elif op == 'MUL':
            if len(stack) < 2:
                raise ValueError(f"line {line_no}: MUL needs two operands")
            a = stack.pop()
            b = stack.pop()
            stack.append(a * b)

        elif op == 'DIV':
            if len(stack) < 2:
                raise ValueError(f"line {line_no}: DIV needs two operands")
            b = stack.pop()
            a = stack.pop()
            if b == 0:
                raise ZeroDivisionError(f"line {line_no}: DIV by zero (divisor = {b})")
            stack.append(a // b)

        elif op == 'DUP':
            if not stack:
                raise ValueError(f"line {line_no}: DUP requires at least one item to duplicate")
            stack.append(stack[-1])

        elif op == 'SWAP':
            if len(stack) < 2:
                raise ValueError(f"line {line_no}: SWAP needs two items, got only {len(stack)} on stack")
            b = stack.pop()
            a = stack.pop()
            stack.append(a)
            stack.append(b)

        elif op == 'PRINT':
            if not stack:
                raise ValueError(f"line {line_no}: PRINT on empty stack")
            outputs.append(str(stack[-1]))

        else:
            # Unknown operation
            raise ValueError(f"line {line_no}: unknown instruction '{op}'")

    return outputs
