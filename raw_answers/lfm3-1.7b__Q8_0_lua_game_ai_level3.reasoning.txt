function BT.Inverter(child)
    local res = {
        tick = function(ctx)
            local childRes = child.tick(ctx)
            if childRes == BT.SUCCESS then
                return BT.FAILURE
            elseif childRes == BT.FAILURE then
                return BT.SUCCESS
            elseif childRes == BT.RUNNING then
                return childRes -- unchanged
            end
        ,
        reset = function()

    }

end
