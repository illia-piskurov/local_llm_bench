        #game-container {

<body>

    <div id="ui">
        <div id="score-board">SCORE: 0</div>
    </div>

    <div id="game-container">
        <canvas id="gameCanvas" width="400" height="400"></canvas>
        <div id="overlay">
            <h1>GAME OVER</h1>
            <p id="final-score"></p>
            <div class="restart-hint">Press <b style="color: #fff;">SPACE</b> to Restart</div>
        </div>
    </div>

    <div class="controls-hint">Arrows / WASD to Move</div>

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score-board');
        const overlay = document.getElementById('overlay');
        const finalScoreElement = document.getElementById('final-score');

        const gridSize = 20;
        const tileCount = canvas.width / gridSize;

        let score = 0;
        let dx = 1;
        let dy = 0;
        let snake = [];
        let food = { x: 15, y: 10 };
        let particles = [];
        let flashOpacity = 0;
        let isGameOver = false;
        let lastUpdateTime = 0;
        let moveInterval = 100; // ms between moves

        function initGame() {
            score = 0;
            dx = 1;
            dy = 0;
            snake = [
                { x: 10, y: 10 },
                { x: 9, y: 10 },
                { x: 8, y: 10 }
            ];
            scoreElement.textContent = `SCORE: ${score}`;
            isGameOver = false;
            overlay.style.display = 'none';
            particles = [];
            flashOpacity = 0;
            createFood();
            requestAnimationFrame(gameLoop);
        }

        function createFood() {
            food = {
                x: Math.floor(Math.random() * tileCount),
                y: Math.floor(Math.random() * tileCount)
            };
            // Prevent food spawning on snake
            for (let segment of snake) {
                if (segment.x === food.x && segment.y === food.y) {
                    createFood();
                    return;
                }
            }
        }

        class Particle {
            constructor(x, y, color) {
                this.x = x;
                this.y = y;
                this.size = Math.random() * 4 + 2;
                this.speedX = (Math.random() - 0.5) * 8;
                this.speedY = (Math.random() - 0.5) * 8;
                this.color = color;
                this.alpha = 1;
            }
            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                this.alpha -= 0.03;
            }
            draw() {
                ctx.save();
                ctx.globalAlpha = this.alpha;
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x, this.y, this.size, this.size);
                ctx.restore();
            }
        }

        function createExplosion(x, y, color) {
            for (let i = 0; i < 12; i++) {
                particles.push(new Particle(x, y, color));
            }
        }

        function update(timestamp) {
            if (isGameOver) return;

            if (timestamp - lastUpdateTime >= moveInterval) {
                lastUpdateTime = timestamp;
                moveSnake();
            }

            // Update particles
            for (let i = particles.length - 1; i >= 0; i--) {
                particles[i].update();
                if (particles[i].alpha <= 0) particles.splice(i, 1);
            }

            // Fade flash effect
            if (flashOpacity > 0) flashOpacity -= 0.05;
        }

        function moveSnake() {
            const head = { x: snake[0].x + dx, y: snake[0].y + dy };

            // Wall collision
            if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {
                return gameOver();
            }

            // Self collision
            for (let i = 0; i < snake.length; i++) {
                if (head.x === snake[i].x && head.y === snake[i].y) {
                    return gameOver();
                }
            }

            snake.unshift(head);

            // Eat food
            if (head.x === food.x && head.y === food.y) {
                score += 10;
                scoreElement.textContent = `SCORE: ${score}`;
                flashOpacity = 0.4;
                createExplosion(food.x * gridSize + gridSize/2, food.y * gridSize + gridSize/2, '#e74c3c');
                createFood();
            } else {
                snake.pop();
            }
        }

        function draw() {
            // Clear canvas
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Draw subtle grid
            ctx.strokeStyle = '#1a1a1a';
            ctx.lineWidth = 1;
            for (let i = 0; i <= tileCount; i++) {
                ctx.beginPath();
                ctx.moveTo(i * gridSize, 0);
                ctx.lineTo(i * gridSize, canvas.height);
                ctx.stroke();
                ctx.beginPath();
                ctx.moveTo(0, i * gridSize);
                ctx.lineTo(canvas.width, i * gridSize);
                ctx.stroke();
            }

            // Draw Food with glow and pulse
            const pulse = Math.sin(Date.now() / 100) * 3;
            ctx.shadowBlur = 15;
            ctx.shadowColor = '#e74c3c';
            ctx.fillStyle = '#e74c3c';
            ctx.beginPath();
            ctx.arc(
                food.x * gridSize + gridSize/2, 
                food.y * gridSize + gridSize/2, 
                (gridSize/2 - 4) + pulse/2, 
                0, Math.PI * 2
            );
            ctx.fill();
            ctx.shadowBlur = 0;

            // Draw Snake
            snake.forEach((segment, index) => {
                const isHead = index === 0;
                const gradient = ctx.createLinearGradient(
                    segment.x * gridSize, 
                    segment.y * gridSize, 
                    (segment.x + 1) * gridSize, 
                    (segment.y + 1) * gridSize
                );
                
                if (isHead) {
                    gradient.addColorStop(0, '#2ecc71');
                    gradient.addColorStop(1, '#27ae60');
                    ctx.shadowBlur = 15;
                    ctx.shadowColor = '#2ecc71';
                } else {
                    gradient.addColorStop(0, '#27ae60');
                    gradient.addColorStop(1, '#1e8449');
                    ctx.shadowBlur = 0;
                }

                ctx.fillStyle = gradient;
                
                // Rounded rectangle for snake segments
                const r = isHead ? 6 : 4;
                const x = segment.x * gridSize + 1;
                const y = segment.y * gridSize + 1;
                const w = gridSize - 2;
                const h = gridSize - 2;
                
                ctx.beginPath();
                ctx.roundRect(x, y, w, h, r);
                ctx.fill();
                ctx.shadowBlur = 0;
            });

            // Draw Particles
            particles.forEach(p => p.draw());

            // Flash effect
            if (flashOpacity > 0) {
                ctx.fillStyle = `rgba(255, 255, 255, ${flashOpacity})`;
                ctx.fillRect(0, 0, canvas.width, canvas.height);
            }
        }

        function gameOver() {
            isGameOver = true;
            finalScoreElement.textContent = `FINAL SCORE: ${score}`;
            overlay.style.display = 'flex';
        }

        function gameLoop(timestamp) {
            update(timestamp);
            draw();
            if (!isGameOver || particles.length > 0) {
                requestAnimationFrame(gameLoop);
            }
        }

        function handleKeyPress(e) {
            const key = e.key;
            const goingUp = dy === -1;
            const goingDown = dy === 1;
            const goingRight = dx === 1;
            const goingLeft = dx === -1;

            if (isGameOver && (key === ' ' || key === 'Enter')) {
                initGame();
                return;
            }

            if ((key === 'ArrowUp' || key === 'w') && !goingDown) {
                dx = 0; dy = -1;
            }
            if ((key === 'ArrowDown' || key === 's') && !goingUp) {
                dx = 0; dy = 1;
            }
            if ((key === 'ArrowLeft' || key === 'a') && !goingRight) {
                dx = -1; dy = 0;
            }
            if ((key === 'ArrowRight' || key === 'd') && !goingLeft) {
                dx = 1; dy = 0;
            }
        }

        window.addEventListener('keydown', handleKeyPress);
        initGame();
    </script>
</body>
</html>
