#include <stdint.h>

#define BUFFER_SIZE 512

static uint8_t buffer[BUFFER_SIZE];
static size_t head = 0;
static size_t tail = 0;

void ringbuf_init(void) {
    head = 0;
    tail = 0;
}

int ringbuf_push(const uint8_t *data, int len) {
    if (ringbuf_available() + len > BUFFER_SIZE) {
        len = BUFFER_SIZE - ringbuf_available();
        if (len <= 0) return -1; // buffer full
    }
    
    for (int i = 0; i < len; ++i) {
        buffer[head] = data[i];
        head = (head + 1) % BUFFER_SIZE;
    }
    return len;
}

void ringbuf_advance(size_t n) {
    if (n >= ringbuf_available()) {
        tail = head;
    } else {
        tail = (tail + n) % BUFFER_SIZE;
    }
}

int ringbuf_available(void) {
    size_t diff = (head >= tail) ? (head - tail) : (BUFFER_SIZE - (tail - head));
    return diff;
}

int ringbuf_free_space(void) {
    return BUFFER_SIZE - ringbuf_available();
}

#include <stdint.h>

#define MAGIC_BYTE 0xAA
#define PAYLOAD_START 3

int get_next_packet(uint8_t *out_payload, int *out_type) {
    while (ringbuf_available() >= 4) { // need at least magic + type + len + checksum
        if (buffer[tail] != MAGIC_BYTE) {
            ringbuf_advance(1);
            continue; // not magic byte, skip to next byte
        }
        
        uint8_t type = buffer[tail + 1];
        uint8_t len = buffer[tail + 2];
        size_t payload_start = tail + PAYLOAD_START;
        
        if (payload_start + len > ringbuf_available()) {
            // incomplete packet, wait for more data
            return -1;
        }
        
        uint8_t checksum = buffer[ringbuf_available() == BUFFER_SIZE ? tail : tail + ringbuf_available() - 1];
        
        uint8_t calculated_checksum = 0;
        for (size_t j = payload_start; j < payload_start + len; ++j) {
            calculated_checksum ^= buffer[j];
        }
        
        if (calculated_checksum != checksum) {
            // invalid checksum, skip to next candidate
            ringbuf_advance(1);
            continue;
        }
        
        // valid packet found, pop it from the ring buffer
        for (size_t j = 0; j < len; ++j) {
            out_payload[j] = buffer[payload_start + j];
        }
        *out_type = type;
        ringbuf_advance(4 + len); // advance head past this packet
        return len;
    }
    return -1; // no complete packet found yet
}
