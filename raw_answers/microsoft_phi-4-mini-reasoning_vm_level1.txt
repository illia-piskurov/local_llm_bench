                n = int(num_str)
                stack.append(n)

                # Do nothing, just pop
            elif instr in ('ADD', 'SUB', 'MUL', 'DIV'):

                b = stack.pop()
                a = stack.pop()
                # Compute result based on operation
                if instr == 'ADD':
                    res = a + b
                elif instr == 'SUB':
                    res = a - b
                elif instr == 'MUL':
                    res = a * b
                else:  # DIV
                    if b ==0:

                    res = a // b
                stack.append(res)
            elif instr == 'DUP':

                output.append(str(stack[-1]))
            else:
                # Unknown instruction, ignore?
                pass
        except Exception as e:
            print(e)
    return output
