def run(program: str) -> list[str]:
    stack = []
    output = []
    lines = program.splitlines()
    for idx, line in enumerate(lines):
        lineno = idx + 1
        stripped = line.strip()
        if not stripped or stripped.startswith("#"):
            continue
        parts = stripped.split()
        op = parts[0].upper()
        try:
            if op == "PUSH":
                if len(parts) != 2:
                    raise ValueError(f"Line {lineno}: PUSH requires exactly one argument")
                stack.append(int(parts[1]))
            elif op == "POP":
                if len(stack) < 1:
                    raise IndexError(f"Line {lineno}: POP on empty stack")
                stack.pop()
            elif op == "ADD":
                if len(stack) < 2:
                    raise IndexError(f"Line {lineno}: ADD needs two values")
                b = stack.pop()
                a = stack.pop()
                stack.append(a + b)
            elif op == "SUB":
                if len(stack) < 2:
                    raise IndexError(f"Line {lineno}: SUB needs two values")
                b = stack.pop()
                a = stack.pop()
                stack.append(a - b)
            elif op == "MUL":
                if len(stack) < 2:
                    raise IndexError(f"Line {lineno}: MUL needs two values")
                b = stack.pop()
                a = stack.pop()
                stack.append(a * b)
            elif op == "DIV":
                if len(stack) < 2:
                    raise IndexError(f"Line {lineno}: DIV needs two values")
                b = stack.pop()
                a = stack.pop()
                if b == 0:
                    raise ZeroDivisionError(f"Line {lineno}: division by zero")
                stack.append(a // b)
            elif op == "DUP":
                if len(stack) < 1:
                    raise IndexError(f"Line {lineno}: DUP on empty stack")
                stack.append(stack[-1])
            elif op == "SWAP":
                if len(stack) < 2:
                    raise IndexError(f"Line {lineno}: SWAP needs two values")
                stack[-1], stack[-2] = stack[-2], stack[-1]
            elif op == "PRINT":
                if len(stack) < 1:
                    raise IndexError(f"Line {lineno}: PRINT on empty stack")
                output.append(str(stack[-1]))
            else:
                raise ValueError(f"Line {lineno}: unknown instruction '{op}'")
        except (IndexError, ZeroDivisionError, ValueError) as e:
            raise e
