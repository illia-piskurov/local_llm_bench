```python
def run(program: str) -> list[str]:
    stack = []                     # each element is a dict for the current transaction
    global_store = {}              # final values after all commits
    outputs = []

    for line in program.splitlines():
        if not line.strip():
            continue
        tokens = line.split()
        cmd = tokens[0]

        if cmd == "SET":
            key, value = tokens[1], tokens[2]
            stack[-1][key] = value

        elif cmd == "GET":
            key = tokens[1]
            val = stack[-1].get(key, "NULL")
            outputs.append(val)

        elif cmd == "DELETE":
            key = tokens[1]
            if key in stack[-1]:
                del stack[-1][key]

        elif cmd == "BEGIN":
            stack.append({})

        elif cmd == "COMMIT":
            if not stack:
                outputs.append("NO TRANSACTION")
                continue
            inner = stack.pop()
            if stack:                     # merge into parent transaction
                outer = stack[-1]
                outer.update(inner)
            else:                          # outermost transaction, effects become global
                global_store.update(inner)

        elif cmd == "ROLLBACK":
            if not stack:
                outputs.append("NO TRANSACTION")
                continue
            inner = stack.pop()           # discard changes

    return outputs
```
