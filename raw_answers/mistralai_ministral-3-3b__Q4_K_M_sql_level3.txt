def compile_query(query: dict) -> dict:
    def _build_where_clause(cond):
        if isinstance(cond, dict) and 'AND' in cond:
            return f"({', '.join(_build_where_clause(c) for c in cond['AND'])})"
        elif isinstance(cond, dict) and 'OR' in cond:
            return f"({', '.join(_build_where_clause(c) for c in cond['OR'])})"
        else:
            field, op, value = cond
            if op == '=':
                return f"{field} = ?"
            elif op == '!=':
                return f"{field} != ?"
            elif op == '>':
                return f"{field} > ?"
            elif op == '<':
                return f"{field} < ?"
            elif op == '>=':
                return f"{field} >= ?"
            elif op == '<=':
                return f"{field} <= ?"
            elif op == 'IN' and isinstance(value, dict):  # Subquery IN
                subquery_sql = compile_subquery(value)
                return f"{field} IN ({subquery_sql})"
            elif op == 'IN':
                placeholders = ', '.join(['?'] * len(value))
                return f"{field} IN ({placeholders})"
            elif op == 'IS NULL':
                return f"{field} IS NULL"
            elif op == 'IS NOT NULL':
                return f"{field} IS NOT NULL"
            elif op == 'LIKE':
                return f"{field} LIKE ?"
            else:
                raise ValueError(f"Unsupported operator: {op}")

    def compile_subquery(subquery):
        sub_params = []
        sub_placeholders = []

        def _traverse_and_collect(cond, params=None, placeholders=None):
            if params is None:
                params = []
            if placeholders is None:
                placeholders = []
            if isinstance(cond, dict) and 'AND' in cond:
                for c in cond['AND']:
                    _traverse_and_collect(c, params, placeholders)
            elif isinstance(cond, dict) and 'OR' in cond:
                for c in cond['OR']:
                    _traverse_and_collect(c, params, placeholders)
            else:
                _, op, value = cond
                if op == 'IN':
                    sub_placeholders.extend(['?'] * len(value))
                elif op == 'LIKE':
                    sub_placeholders.append('?')
                    sub_params.append(value)
                else:
                    sub_placeholders.append('?')
                    sub_params.append(value)

        _traverse_and_collect(subquery['where'], params, placeholders=sub_placeholders)

        # Build SELECT clause for subquery
        select_clause = "SELECT "
        if subquery.get('select') == ['*']:
            select_clause += "*"
        else:
            select_clause += ", ".join(subquery['select'])

        # Build WHERE clause with parameter placeholders
        where_placeholders = []
        _traverse_and_collect(subquery['where'], [], placeholders=where_placeholders)
        where_sql = _build_where_clause(subquery['where'])
        if where_sql:
            select_clause += " WHERE " + where_sql

        # Build JOIN clause
        join_clause = _build_join_clause(subquery.get('joins', []))

        # Build GROUP BY clause
        group_by_clause = _build_group_by_clause(subquery.get('groupBy', []))

        sql = f"{select_clause} FROM {subquery['table']}"
        if join_clause:
            sql += join_clause
        sql += group_by_clause

        return sql, sub_params

    def compile_having(having_cond):
        if isinstance(having_cond, dict) and 'AND' in having_cond:
            clauses = [_build_having_clause(c) for c in having_cond['AND']]
            return f"HAVING {', '.join(clauses)}"
        elif isinstance(having_cond, dict) and 'OR' in having_cond:
            clauses = [_build_having_clause(c) for c in having_cond['OR']]
            return f"HAVING {', '.join(clauses)}"
        else:
            field, op, value = having_cond
            if op == '=':
                return f"{field} = ?"
            elif op == '>':
                return f"{field} > ?"
            elif op == '<':
                return f"{field} < ?"
            elif op == '>=':
                return f"{field} >= ?"
            elif op == '<=':
                return f"{field} <= ?"
            elif op == 'IN':
                placeholders = ', '.join(['?'] * len(value))
                return f"{field} IN ({placeholders})"
            elif op == 'IS NULL':
                return f"{field} IS NULL"
            elif op == 'IS NOT NULL':
                return f"{field} IS NOT NULL"
            else:
                raise ValueError(f"Unsupported operator: {op}")

    def _build_having_clause(cond):
        if isinstance(cond, dict) and 'AND' in cond:
            return f"({', '.join(_build_having_clause(c) for c in cond['AND'])})"
        elif isinstance(cond, dict) and 'OR' in cond:
            return f"({', '.join(_build_having_clause(c) for c in cond['OR'])})"
        else:
            field, op, value = cond
            if op == '=':
                return f"{field} = ?"
            elif op == '>':
                return f"{field} > ?"
            elif op == '<':
                return f"{field} < ?"
            elif op == '>=':
                return f"{field} >= ?"
            elif op == '<=':
                return f"{field} <= ?"
            elif op == 'IN':
                placeholders = ', '.join(['?'] * len(value))
                return f"{field} IN ({placeholders})"
            elif op == 'IS NULL':
                return f"{field} IS NULL"
            elif op == 'IS NOT NULL':
                return f"{field} IS NOT NULL"
            elif isinstance(value, dict):  # Subquery IN
                subquery_sql = compile_subquery(value)[0]
                return f"{field} IN ({subquery_sql})"
            else:
                raise ValueError(f"Unsupported operator: {op}")

    def _build_select_clause(select_items):
        clauses = []
        params_count = 0

        for item in select_items:
            if isinstance(item, dict) and 'expr' in item:
                expr = item['expr']
                alias = item.get('as', '')
                # Handle computed expressions like COUNT()
                if expr.startswith(('COUNT(', 'SUM(', 'AVG(', 'MAX(', 'MIN(')):
                    clauses.append(f"{expr[:-1]} AS {alias}")
                else:
                    clauses.append(f"{expr} AS {alias}")
            else:
                clauses.append(item)

        return ", ".join(clauses)

    def _build_where_clause_extended(cond, params=None):
        if params is None:
            params = []
        placeholders = []

        if isinstance(cond, dict) and 'AND' in cond:
            for c in cond['AND']:
                _traverse_and_collect(c, params, placeholders)
        elif isinstance(cond, dict) and 'OR' in cond:
            for c in cond['OR']:
                _traverse_and_collect(c, params, placeholders)
        else:
            field, op, value = cond
            if op == '=':
                placeholders.append('?')
                params.append(value)
            elif op == '!=':
                placeholders.append('?')
                params.append(value)
            elif op == '>':
                placeholders.append('?')
                params.append(value)
            elif op == '<':
                placeholders.append('?')
                params.append(value)
            elif op == '>=':
                placeholders.append('?')
                params.append(value)
            elif op == '<=':
                placeholders.append('?')
                params.append(value)
            elif op == 'IN' and isinstance(value, dict):  # Subquery IN
                subquery_sql, sub_params = compile_subquery(value)
                placeholders.extend(sub_params)
                return f"{field} IN ({subquery_sql})"
            elif op == 'IN':
                placeholders.extend(['?'] * len(value))
            elif op == 'IS NULL':
                return f"{field} IS NULL"
            elif op == 'IS NOT NULL':
                return f"{field} IS NOT NULL"
            elif op == 'LIKE':
                placeholders.append('?')
                params.append(value)
            else:
                raise ValueError(f"Unsupported operator: {op}")

        return placeholders

    def _traverse_and_collect(cond, params=None, placeholders=None):
        if params is None:
            params = []
        if placeholders is None:
            placeholders = []

        if isinstance(cond, dict) and 'AND' in cond:
            for c in cond['AND']:
                _traverse_and_collect(c, params, placeholders)
        elif isinstance(cond, dict) and 'OR' in cond:
            for c in cond['OR']:
                _traverse_and_collect(c, params, placeholders)
        else:
            _, op, value = cond
            if op == 'IN':
                placeholders.extend(['?'] * len(value))
            elif op == 'LIKE':
                placeholders.append('?')
                params.append(value)
            elif isinstance(value, dict):  # Subquery IN
                subquery_sql, sub_params = compile_subquery(value)
                placeholders.extend(sub_params)
            else:
                placeholders.append('?')
                params.append(value)

    def _build_join_clause(joins):
        if not joins:
            return ""
        clauses = []
        for join in joins:
            join_type = join.get('type', 'INNER')
            table = join['table']
            on_parts = []
            for left_col, right_col in join['on'].items():
                on_parts.append(f"{left_col} = {right_col}")
            clauses.append(f"JOIN {join_type} {table} ON {' AND '.join(on_parts)}")
        return " ".join(clauses)

    def _build_group_by_clause(group_columns):
        if not group_columns:
            return ""
        return f"GROUP BY {', '.join(group_columns)}"

    def _build_having_clause_extended(having_cond, params=None):
        if params is None:
            params = []
        placeholders = []

        if isinstance(having_cond, dict) and 'AND' in having_cond:
            for c in having_cond['AND']:
                _traverse_and_collect(c, params, placeholders)
        elif isinstance(having_cond, dict) and 'OR' in having_cond:
            for c in having_cond['OR']:
                _traverse_and_collect(c, params, placeholders)
        else:
            field, op, value = having_cond
            if op == '=':
                placeholders.append('?')
                params.append(value)
            elif op == '>':
                placeholders.append('?')
                params.append(value)
            elif op == '<':
                placeholders.append('?')
                params.append(value)
            elif op == '>=':
                placeholders.append('?')
                params.append(value)
            elif op == '<=':
                placeholders.append('?')
                params.append(value)
            elif op == 'IN' and isinstance(value, dict):  # Subquery IN
                subquery_sql, sub_params = compile_subquery(value)
                placeholders.extend(sub_params)
            elif op == 'IN':
                placeholders.extend(['?'] * len(value))
            elif op == 'IS NULL':
                return f"{field} IS NULL"
            elif op == 'IS NOT NULL':
                return f"{field} IS NOT NULL"
            else:
                raise ValueError(f"Unsupported operator: {op}")

        return placeholders

    # Build SELECT clause with computed expressions
    select_clause = "SELECT "
    if query.get('select') == ['*']:
        select_clause += "*"
    else:
        select_items = []
        for item in query['select']:
            if isinstance(item, dict) and 'expr' in item:
                select_items.append(f"{item['expr']} AS {item.get('as', '')}")
            else:
                select_items.append(item)
        select_clause += ", ".join(select_items)

    # Build WHERE clause with parameter placeholders
    where_placeholders = []
    params = []
    if 'where' in query and query['where'] is not None:
        _traverse_and_collect(query['where'], params, where_placeholders)
        where_clause = _build_where_clause_extended(query['where'])
        where_sql = _build_where_clause(query['where'])

    # Build JOIN clause
    join_clause = _build_join_clause(query.get('joins', []))

    # Build GROUP BY clause
    group_by_clause = _build_group_by_clause(query.get('groupBy', []))

    # Build HAVING clause with parameter placeholders
    having_placeholders = []
    params.extend([p for p in query.get('having', {}).get('params', [])])
    if 'having' in query and query['having'] is not None:
        _traverse_and_collect(query['having'], params, having_placeholders)
        having_sql = compile_having(query['having'])

    # Build SQL query
    sql = f"{select_clause} FROM {query['table']}"
    if join_clause:
        sql += join_clause
    if where_sql and 'where' in query and query['where']:
        sql += " WHERE " + where_sql
    if group_by_clause:
        sql += group_by_clause
    if having_sql and 'having' in query and query['having'] is not None:
        sql += " HAVING " + having_sql

    return {"sql": sql, "params": params}
