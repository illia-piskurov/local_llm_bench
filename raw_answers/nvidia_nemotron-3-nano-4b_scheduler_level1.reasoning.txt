def topo_sort(tasks):
    # Build adjacency list and indegree
    indeg = {t:0 for t in tasks}
    graph = {}
    for task, deps in tasks.items():
        graph[task] = []
        for d in deps:
            if d not in tasks: maybe ignore? but assume valid.
            indeg[task] += 1
            # also add edge from dep to task
            graph.setdefault(d, []).append(task)
    # Kahn's algorithm
    queue = [t for t in tasks if indeg[t]==0]
    order=[]
    while queue:
        u=queue.pop(0) # arbitrary order
        order.append(u)
        for v in graph.get(u,[]):
            indeg[v]-=1
            if indeg[v]==0:
                queue.append(v)
    if len(order)==len(tasks):
        return order
    else:
        return None
