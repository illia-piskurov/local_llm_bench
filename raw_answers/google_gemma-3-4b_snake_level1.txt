```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Snake Game</title>
    <style>
        body {
            margin: 0;
            overflow: hidden;
            background-color: #2d3e59;
        }
        canvas {
            display: block;
        }
    </style>
</head>
<body>

    <canvas id="gameCanvas" width="400" height="400"></canvas>

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        let scale = 10;
        let snakeX = [];
        let snakeY = [];
        let foodX, foodY;
        let direction = 'right';
        let score = 0;
        let gameInterval;

        function init() {
            snakeX = [20, 20, 20];
            snakeY = [20, 20, 20];
            foodX = Math.floor(Math.random() * 40) * scale;
            foodY = Math.floor(Math.random() * 40) * scale;
            direction = 'right';
            score = 0;

            draw();
        }

        function draw() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.scale(scale, scale);

            for (let i = 0; i < snakeX.length; i++) {
                ctx.fillStyle = 'green';
                ctx.fillRect(snakeX[i], snakeY[i], scale, scale);
            }

            ctx.fillStyle = 'red';
            ctx.font = "20px Arial";
            ctx.fillText("Score: " + score, 10, 20);
        }


        function update() {
            let headX = snakeX[0];
            let headY = snakeY[0];

            if (direction === 'up') {
                headY -= scale;
            } else if (direction === 'down') {
                headY += scale;
            } else if (direction === 'left') {
                headX -= scale;
            } else if (direction === 'right') {
                headX += scale;
            }

            if (headX < 0 || headX >= canvas.width / scale || headY < 0 || headY >= canvas.height / scale) {
                clearInterval(gameInterval);
                alert("Game Over! Score: " + score);
                init();
            }

            for (let i = 0; i < snakeX.length; i++) {
                if (snakeX[i] === headX && snakeY[i] === headY) {
                    clearInterval(gameInterval);
                    alert("Game Over! Score: " + score);
                    init();
                }
            }

            if (headX === foodX && headY === foodY) {
                score++;
                foodX = Math.floor(Math.random() * 40) * scale;
                foodY = Math.floor(Math.random() * 40) * scale;
            }

            snakeX.unshift(headX);
            snakeY.unshift(headY);
        }


        document.addEventListener('keydown', function (e) {
            if (e.key === 'ArrowUp') {
                direction = 'up';
            } else if (e.key === 'ArrowDown') {
                direction = 'down';
            } else if (e.key === 'ArrowLeft') {
                direction = 'left';
            } else if (e.key === 'ArrowRight') {
                direction = 'right';
            }
        });

        init();
        gameInterval = setInterval(update, 100);
    </script>

</body>
</html>
```
