from collections import deque

def topo_sort(tasks):
    # Собираем все узлы: ключи словаря + уникальные значения (dependencies)
    nodes = set()
    for task in tasks:
        deps_set = set(tasks[task])
        nodes.add(task)
        nodes |= deps_set
    
    graph = {node: [] for node in nodes}
    indegree = {node: 0 for node in nodes}

    # Заполняем граф и входные степени
    for task, deps_list in tasks.items():
        deps_set = set(deps_list)   # Уникализируем зависимости для этой задачи (для одного словаря)
        for dep in deps_set:
            graph[dep].append(task)
            indegree[task] += 1

    queue = deque([node for node in nodes if indegree[node] == 0])
    topo_order = []

    while queue:
        node = queue.popleft()
        topo_order.append(node)

        # Обновляем входные степени для всех задач, которые зависят от node
        for neighbor in graph[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                queue.append(neighbor)

    if len(topo_order) != len(nodes):
        return None

    return topo_order
