def compile_query(query: dict) -> dict:
    def build_condition(condition):
        if isinstance(condition, dict):
            if 'AND' in condition:
                return '(' + ' AND '.join(build_condition(c) for c in condition['AND']) + ')'
            elif 'OR' in condition:
                return '(' + ' OR '.join(build_condition(c) for c in condition['OR']) + ')'
            else:
                field = list(condition.keys())[0]
                op = condition[field]
                if isinstance(op, str):
                    return f"{field} {op}"
                value = op
                params.append(value)
                return "$" + str(len(params))
        elif isinstance(condition, list):
            return ' AND '.join(build_condition(c) for c in condition)

    def build_select(columns):
        return "SELECT " + (",".join(columns) if columns else "*")

    def build_order_by(order_by):
        order_clauses = [f"{field} {direction}" for field, direction in (o.get('field', ''), o.get('dir', 'ASC'))]
        return "ORDER BY " + ", ".join(order_clauses) if order_by else ""

    def build_limit_offset(limit, offset):
        limit_clause = f"LIMIT {limit}" if limit is not None else ""
        offset_clause = f"OFFSET {offset}" if offset is not None else ""
        return " ".join(filter(None, [limit_clause, offset_clause]))

    params = []
    sql_parts = []

    table = query.get('table')
    sql_parts.append(f"{table}")

    select = query.get('select', ['*'])
    sql_parts.append(build_select(select))

    where = query.get('where')
    if where:
        condition_str = build_condition(where)
        sql_parts.append("WHERE")
        sql_parts.append(condition_str)

    order_by = query.get('orderBy', [])
    sql_parts.append(build_order_by(order_by))

    limit_offset = query.get('limit') or query.get('offset')
    sql_parts.append(build_limit_offset(limit_offset[0] if 'limit' in limit_offset else None, limit_offset[1] if 'offset' in limit_offset else None))

    return {"sql": " ".join(sql_parts), "params": params}
