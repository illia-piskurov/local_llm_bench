def compile_query(query: dict) -> dict:
    def _build_where_clause(cond):
        if isinstance(cond, dict) and 'AND' in cond:
            return f"({', '.join(_build_where_clause(c) for c in cond['AND'])})"
        elif isinstance(cond, dict) and 'OR' in cond:
            return f"({', '.join(_build_where_clause(c) for c in cond['OR'])})"
        else:
            field, op, value = cond
            if op == '=':
                return f"{field} = ?"
            elif op == '!=':
                return f"{field} != ?"
            elif op == '>':
                return f"{field} > ?"
            elif op == '<':
                return f"{field} < ?"
            elif op == '>=':
                return f"{field} >= ?"
            elif op == '<=':
                return f"{field} <= ?"
            elif op == 'IN':
                placeholders = ', '.join(['?'] * len(value))
                return f"{field} IN ({placeholders})"
            elif op == 'IS NULL':
                return f"{field} IS NULL"
            elif op == 'IS NOT NULL':
                return f"{field} IS NOT NULL"
            elif op == 'LIKE':
                return f"{field} LIKE ?"
            else:
                raise ValueError(f"Unsupported operator: {op}")

    def _build_join_clause(joins):
        if not joins:
            return ""
        clauses = []
        for join in joins:
            join_type = join.get('type', 'INNER')
            table = join['table']
            on_parts = []
            for left_col, right_col in join['on'].items():
                on_parts.append(f"{left_col} = {right_col}")
            clauses.append(f"JOIN {join_type} {table} ON {' AND '.join(on_parts)}")
        return " ".join(clauses)

    def _build_group_by_clause(group_columns):
        if not group_columns:
            return ""
        return f"GROUP BY {', '.join(group_columns)}"

    def _traverse_and_collect(cond, placeholders=None):
        if placeholders is None:
            placeholders = []
        params = []

        if isinstance(cond, dict) and 'AND' in cond:
            for c in cond['AND']:
                _traverse_and_collect(c, placeholders)
        elif isinstance(cond, dict) and 'OR' in cond:
            for c in cond['OR']:
                _traverse_and_collect(c, placeholders)
        else:
            _, op, value = cond
            if op == 'IN':
                placeholders.extend(['?'] * len(value))
            elif op == 'LIKE':
                placeholders.append('?')
                params.append(value)
            else:
                placeholders.append('?')
                params.append(value)

        return placeholders

    # Build SELECT clause
    select_clause = "SELECT "
    if query.get('select') == ['*']:
        select_clause += "*"
    else:
        select_clause += ", ".join(query['select'])

    # Build WHERE clause with parameter placeholders and params collection
    where_placeholders = []
    params = []
    if 'where' in query and query['where'] is not None:
        _traverse_and_collect(query['where'], where_placeholders)
        where_clause = _build_where_clause(query['where'])
    else:
        where_clause = ""

    # Build JOIN clause
    join_clause = _build_join_clause(query.get('joins', []))

    # Build GROUP BY clause
    group_by_clause = _build_group_by_clause(query.get('groupBy', []))

    # Build SQL query
    sql = f"{select_clause} FROM {query['table']}"
    if join_clause:
        sql += join_clause
    if where_clause:
        sql += " WHERE " + where_clause
    sql += group_by_clause
    sql += _build_order_by_clause(query.get('orderBy', []))
    sql += _build_limit_offset()

    return {"sql": sql, "params": params}

def _build_order_by_clause(order_conditions):
    if not order_conditions:
        return ""
    clauses = []
    for cond in order_conditions:
        field, dir_ = cond.get('field'), cond.get('dir', 'ASC')
        clauses.append(f"{field} {'DESC' if dir_ == 'DESC' else 'ASC'}")
    return " ORDER BY " + ", ".join(clauses)

def _build_limit_offset():
    limit = query.get('limit', None)
    offset = query.get('offset', None)
    parts = []
    if limit is not None:
        parts.append(f"LIMIT {limit}")
    if offset is not None:
        parts.append(f"OFFSET {offset}")
    return " ".join(parts) if parts else ""
