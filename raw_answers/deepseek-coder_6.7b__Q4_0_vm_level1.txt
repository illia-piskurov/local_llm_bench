class StackMachine:
    def __init__(self):
        self.stack = []
        self.commands = {
            'PUSH': self.push,
            'POP': self.pop,
            'ADD': self.add,
            'SUB': self.sub,
            'MUL': self.mul,
            'DIV': self.div,
            'DUP': self.dup,
            'SWAP': self.swap,
            'PRINT': self.print_top
        }
        self.output = []

    def push(self, value):
        self.stack.append(int(value))

    def pop(self):
        return self.stack.pop()

    def add(self):
        self.stack.append(self.stack.pop() + self.stack.pop())

    def sub(self):
        b = self.stack.pop()
        a = self.stack.pop()
        self.stack.append(a - b)

    def mul(self):
        self.stack.append(self.stack.pop() * self.stack.pop())

    def div(self):
        b = self.stack.pop()
        a = self.stack.pop()
        if b == 0:
            raise Exception('Division by zero at line {0}'.format(line_num))
        self.stack.append(a // b)

    def dup(self):
        self.stack.append(self.stack[-1])

    def swap(self):
        a = self.stack.pop()
        b = self.stack.pop()
        self.stack.append(a)
        self.stack.append(b)

    def print_top(self):
        self.output.append(str(self.stack[-1]))

    def run(self, program):
        lines = program.split('\n')
        for line_num, line in enumerate(lines, start=1):
            if line and not line.startswith('#'):
                instruction, *args = line.split()
                command = self.commands.get(instruction)
                if command:
                    if args:
                        command(*args)
                    else:
                        command()
                else:
                    raise Exception('Unknown instruction at line {0}'.format(line_num))
        return self.output
