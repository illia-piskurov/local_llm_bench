int decode_packet(const uint8_t *stream, int stream_len, uint8_t *out_payload, int *out_type)
{
    if (!stream || !out_payload || !out_type) return -1; // maybe not needed but safe.

    for (int i = 0; i + 4 <= stream_len; ++i) {
        if (stream[i] != 0xAA) continue;
        uint8_t type = stream[i+1];
        uint8_t len = stream[i+2];
        // ensure payload and checksum exist
        if (i + len + 3 >= stream_len) continue;

        int xor_val = 0;
        for (int j = 0; j < len; ++j)
            xor_val ^= stream[i+3+j];

        if (xor_val == stream[i+len+3]) {
            // valid packet
            *out_type = type;
            for (int j = 0; j < len; ++j)
                out_payload[j] = stream[i+3+j];
            return len;
        }
    }
    return -1;
}
