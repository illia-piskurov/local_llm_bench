#include <stdint.h>

#define BUFFER_SIZE 512

static uint8_t buffer[BUFFER_SIZE];
static size_t head = 0;
static size_t tail = 0;

void ringbuf_init(void) {
    head = 0;
    tail = 0;
}

int ringbuf_push(uint8_t byte) {
    if (ringbuf_available() == BUFFER_SIZE) {
        return -1; /* buffer full */
    }
    buffer[head] = byte;
    head = (head + 1) % BUFFER_SIZE;
    return 0;
}

int ringbuf_pop(void) {
    if (tail == head) {
        return -1; /* buffer empty */
    }
    int value = buffer[tail];
    tail = (tail + 1) % BUFFER_SIZE;
    return value;
}

int ringbuf_available(void) {
    size_t diff = (head >= tail) ? (head - tail) : (BUFFER_SIZE - (tail - head));
    return diff;
}

int ringbuf_free_space(void) {
    return BUFFER_SIZE - ringbuf_available();
}

#include <stdint.h>

#define MAGIC_BYTE 0xAA
#define PAYLOAD_START 3

int decode_packet(const uint8_t *stream, int stream_len, uint8_t *out_payload, int *out_type) {
    size_t i = 0;
    while (i + 4 <= stream_len) { // need at least magic + type + len + checksum
        if (stream[i] != MAGIC_BYTE) {
            i++;
            continue; // not magic byte, skip to next byte
        }
        uint8_t type = stream[i + 1];
        uint8_t len = stream[i + 2];
        size_t payload_start = i + PAYLOAD_START;
        
        if (payload_start + len >= stream_len) {
            i++; // not enough data for the length, skip to next byte
            continue;
        }
        uint8_t checksum = stream[stream_len - 1]; // last byte is checksum
        
        uint8_t calculated_checksum = 0;
        for (size_t j = payload_start; j < payload_start + len; ++j) {
            calculated_checksum ^= stream[j];
        }
        
        if (calculated_checksum != checksum) {
            i++; // invalid checksum, skip to next byte
            continue;
        }
        
        if (out_payload) {
            for (size_t j = 0; j < len; ++j) {
                out_payload[j] = stream[payload_start + j];
            }
        }
        *out_type = type;
        return len;
    }
    return -1; // no valid packet found
}
