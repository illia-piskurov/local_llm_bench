def critical_path(tasks):
    if not tasks: return 0
    # Build indegree and adjacency
    indeg = {}
    for task in tasks:
        indeg[task]=0
    for deps in tasks.values():
        for d in deps:
            indeg[d] = indeg.get(d,0)+1

    zero_indeg = deque([t for t,d in indeg.items() if d==0])
    order=[]
    while zero_indeg:
        node=zero_indeg.popleft()
        order.append(node)
        for succ,deps in tasks.items():
            if node in deps:
                indeg[succ] -=1
                if indeg[succ]==0:
                    zero_indeg.append(succ)

    # If cycle:
    if len(order)!=len(tasks):
        return None

    # compute critical path length
    finish = {}
    for node in order:
        pred_finish= max(finish[p] for p in tasks[node][1]) if tasks[node][1] else 0
        finish[node]=tasks[node][0]+pred_finish
    return max(finish.values()) if finish else 0
