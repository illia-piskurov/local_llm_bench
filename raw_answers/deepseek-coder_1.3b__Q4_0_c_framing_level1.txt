Here is the C code for the fixed-size ring buffer:

```c
#include <stdint.h>

#define BUFFER_SIZE 512

static uint8_t buffer[BUFFER_SIZE];
static int head = 0;
static int tail = 0;

void ringbuf_init(void) {
    head = tail = 0;
}

int ringbuf_push(uint8_t byte) {
    if ((head + 1) % BUFFER_SIZE == tail) {
        return -1;
    }
    buffer[head] = byte;
    head = (head + 1) % BUFFER_SIZE;
    return 0;
}

int ringbuf_pop(void) {
    if (head == tail) {
        return -1;
    }
    uint8_t byte = buffer[tail];
    tail = (tail + 1) % BUFFER_SIZE;
    return byte;
}

int ringbuf_available(void) {
    if (head >= tail) {
        return head - tail;
    } else {
        return BUFFER_SIZE - tail + head;
    }
}

int ringbuf_free_space(void) {
    return BUFFER_SIZE - ringbuf_available();
}
```

This code uses a circular buffer with a head and tail pointer. The head pointer points to the next byte to be pushed, and the tail pointer points to the previous byte to be popped. The `ringbuf_push` function increments the head pointer, and the `ringbuf_pop` function increments the tail pointer. The `ringbuf_available` function returns the number of bytes in the buffer, and the `ringbuf_free_space` function returns the remaining free space in the buffer.
