```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Snake Game</title>
    <style>
        body { margin: 0; overflow: hidden; }
        canvas { background:#222; display:block; }
    </style>
</head>
<body onkeydown="moveSnake(event)">
    <canvas id="game" width="400" height="400"></canvas>
    <div id="score">Score: 0</div>
    <button id="restart">Restart</button>
    <script>
        const canvas = document.getElementById('game');
        const ctx = canvas.getContext('2d');
        let snake = [{x: 50, y: 50}];
        let food = {x: Math.round(Math.random()*3)*50+50, y: Math.round(Math.random()*3)*50+50};
        let direction = 'RIGHT';
        let score = 0;
        const cellSize = 20;

        function drawSnake() {
            ctx.fillStyle = 'white';
            snake.forEach(segment => {
                ctx.fillRect(segment.x * cellSize, segment.y * cellSize, cellSize, cellSize);
            });
        }

        function drawFood() {
            ctx.fillStyle = 'green';
            ctx.fillRect(food.x * cellSize - 2, food.y * cellSize - 2, 18, 18);
        }

        function collisionDetect() {
            if (snake.some(segment => segment.x === food.x && segment.y === food.y)) {
                growSnake();
            } else if (snake[0].x < 0 || snake[0].x >= canvas.width / cellSize ||
                       snake[0].y < 0 || snake[0].y >= canvas.height / cellSize) {
                gameover();
            }
        }

        function moveSnake(event) {
            const key = event.keyCode;
            if (key === 37 && direction !== 'RIGHT') direction = 'LEFT';
            else if (key === 38 && direction !== 'DOWN') direction = 'UP';
            else if (key === 39 && direction !== 'LEFT') direction = 'RIGHT';
            else if (key === 40 && direction !== 'UP') direction = 'DOWN';

            const newHead = {x: snake[0].x, y: snake[0].y};
            switch(direction) {
                case 'LEFT': newHead.x -= 1; break;
                case 'RIGHT': newHead.x += 1; break;
                case 'UP': newHead.y -= 1; break;
                case 'DOWN': newHead.y += 1; break;
            }

            collisionDetect();
        }

        function growSnake() {
            snake.unshift({x: snake[0].x - cellSize, y: snake[0].y});
            score++;
            document.getElementById('score').textContent = `Score: ${score}`;
        }

        function gameover() {
            alert('Game Over! Your score was ' + score);
            location.reload();
        }

        function update() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            if (snake[0].x === food.x && snake[0].y === food.y) {
                drawFood();
            } else {
                snake.pop();
            }

            drawSnake();
            drawFood();

            requestAnimationFrame(update);
        }

        function init() {
            update();
        }

        document.getElementById('restart').onclick = () => {
            score = 0;
            document.getElementById('score').textContent = `Score: ${score}`;
            location.reload();
        };

        init();
    </script>
</body>
</html>
```
