def run(program: str) -> List[str]:
    stack = []
    lines = program.splitlines()
    labels = {}
    i = 0
    output = []
    while i < len(lines):
        raw = lines[i].strip()
        if not raw or raw.startswith('#'):
            i += 1
            continue
        parts = raw.split()
        op = parts[0].upper()
        name = None
        arg = ''
        # handle unary with arg, binary none, label/jump etc
        if op == 'LABEL':
            name = parts[1]
            labels[name] = i
            i += 1
            continue
        if op in ['JMP','JZ','JNZ']:
            name = parts[1]
            target_line = labels.get(name)
            if target_line is None:
                raise ValueError(f"Line {i}: Label '{name}' not defined")
            i = target_line + 1   # skip to after label? Actually we want to execute its instructions, so set i = target_line
        elif op == 'PUSH':
            stack.append(int(parts[1]))
            i += 1
        elif op == 'POP':
            if not stack:
                raise ValueError(f"Line {i}: Stack underflow (cannot POP)")
            stack.pop()
            i += 1
        elif op in ['ADD','SUB','MUL','DIV']:
            b = stack.pop(); a = stack.pop()
            if op == 'DIV':
                if b == 0:
                    raise ZeroDivisionError(f"Line {i}: Division by zero")
                result = a // b
            else:
                result = (a + b) if op=='ADD' else (a - b) if op=='SUB' else (a * b)
            stack.append(result)
            i += 1
        elif op == 'DUP':
            if not stack:
                raise ValueError(f"Line {i}: Stack underflow (cannot DUP)")
            stack.append(stack[-1])
            i += 1
        elif op == 'SWAP':
            if len(stack) < 2:
                raise ValueError(f"Line {i}: Not enough values to SWAP")
            a, b = stack.pop(), stack.pop()
            stack.append(a); stack.append(b)
            i += 1
        elif op == 'PRINT':
            output.append(str(stack[-1]))
            i += 1
        else:
            raise ValueError(f"Line {i}: Unknown instruction '{op}'")
    return output
