Key design:
- Worker pool of `concurrency` workers, each pulls next index.
- Each task: attempt with retries/backoff/timeout.
- On task failure: if retries remain, retry; else record settled result.
- On signal abort: reject immediately.
- settled mode: record results as settled objects.

Implementation:

```javascript
async function pMap(items, mapper, options) {
  const opts = ...;
  const concurrency = opts.concurrency ?? 1;
  const retries = opts.retries ?? 0;
  const backoffMs = opts.backoffMs ?? 0;
  const timeoutMs = opts.timeoutMs ?? 0;
  const signal = opts.signal;

  if (signal && signal.aborted) {
    const reason = signal.reason ?? new Error('Aborted');
    return Promise.reject(reason);
  }

  if (items.length === 0) return [];

  const results = new Array(items.length);
  let pending = 0;
  let nextIndex = 0;

  async function run() {
    while (pending < concurrency) {
      if (nextIndex >= items.length) return;
      const index = nextIndex++;
      pending++;
      try {
        results[index] = await runAttempt(index);
      } catch (error) {
        if (settled) results[index] = { status: 'rejected', reason: error };
        else throw error;
      }
      pending--;
    }
  }
  await Promise.all(Array.from({length: concurrency}, run));
  return settled ? results.map(...) : results;
}
```
