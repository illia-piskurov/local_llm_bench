from typing import Optional

def _topological_order(tasks: dict[str, tuple[int, list[str], int]]) -> list[str] | None:
    graph = {name: list(deps) for name, (_, deps, _) in tasks.items()}
    in_degree = {name: 0 for name in tasks}
    for name, deps in graph.items():
        for dep in deps:
            if dep not in tasks:
                return None
            in_degree[name] += 1

    queue = [name for name in tasks if in_degree[name] == 0]
    order = []
    while queue:
        name = queue.pop(0)
        order.append(name)
        for dep in graph[name]:
            in_degree[dep] -= 1
            if in_degree[dep] == 0:
                queue.append(dep)

    if len(order) != len(tasks):
        return None
    return order


def critical_path(tasks: dict[str, tuple[int, list[str], int]]) -> int | None:
    order = _topological_order(tasks)
    if order is None:
        return None

    longest = {name: tasks[name][0] for name in tasks}
    for name in order:
        for dep in graph[name]:
            if longest[dep] + tasks[name][0] > longest[name]:
                longest[name] = longest[dep] + tasks[name][0]

    return max(longest.values()) if longest else 0


def makespan(tasks: dict[str, tuple[int, list[str], int]], workers: int) -> int | None:
    if workers <= 0:
        return None
    if not tasks:
        return 0

    order = _topological_order(tasks)
    if order is None:
        return None

    graph = {name: list(deps) for name, (_, deps, _) in tasks.items()}
    ready = [name for name in tasks if not graph[name]]
    heap = [(-priority, duration, name) for name, (_, _, priority) in tasks.items()]
    import heapq
    heapq.heapify(heap)

    running: dict[str, tuple[int, int]] = {}
    result: list[str] = []
    time = 0

    while running:
        for _ in range(workers):
            if not heap:
                break
            heapq.heappop(heap)

        for _, duration, name in sorted(
            (heapq.heappop(heap) for _ in range(workers)),
            key=lambda item: (-item[0], item[1], item[2]),
        ):
            running[name] = (time, duration)
            result.append(name)

        time += min(duration for _, duration in running.values())

        finished = [name for name, (start, duration) in running.items() if start + duration <= time]
        for name in finished:
            duration, priority = tasks[name][0], tasks[name][2]
            del running[name]
            heapq.heappush(heap, (-priority, duration, name))

    return time
