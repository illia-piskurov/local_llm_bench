#define BUFFER_SIZE 512

typedef struct {
    uint8_t data[BUFFER_SIZE];
    size_t head;
    size_t tail;
} RingBuffer;

static RingBuffer ringbuf = {
    .head = 0,
    .tail = 0
};

void ringbuf_init(void) {
    ringbuf.head = 0;
    ringbuf.tail = 0;
}

int ringbuf_push(uint8_t byte) {
    if (ringbuf.tail == ringbuf.head && ringbuf.head == BUFFER_SIZE) {
        return -1; // full
    }
    ringbuf.data[ringbuf.tail] = byte;
    ringbuf.tail = (ringbuf.tail + 1) % BUFFER_SIZE;
    return 0;
}

int ringbuf_pop(void) {
    if (ringbuf.head == ringbuf.tail) {
        return -1; // empty
    }
    uint8_t byte = ringbuf.data[ringbuf.head];
    ringbuf.head = (ringbuf.head + 1) % BUFFER_SIZE;
    return byte;
}

int ringbuf_available(void) {
    return (ringbuf.tail - ringbuf.head + BUFFER_SIZE) % BUFFER_SIZE;
}

int ringbuf_free_space(void) {
    return (BUFFER_SIZE - ringbuf_available());
}

/* Packet decoder */
int decode_packet(const uint8_t *stream, int stream_len, uint8_t *out_payload, int *out_type) {
    if (stream_len < 4) return -1;
    size_t i = 0;
    while (i < stream_len) {
        if (stream[i] == 0xAA) {
            if (i + 4 > stream_len) return -1;
            uint8_t magic = stream[i];
            uint8_t type = stream[i + 1];
            uint8_t len = stream[i + 2];
            uint8_t checksum = stream[i + 3];
            if (len > 255 || i + 4 + len > stream_len) return -1;
            uint8_t expected_checksum = 0;
            for (int j = i + 1; j < i + 1 + len; ++j) {
                expected_checksum ^= stream[j];
            }
            if (checksum != expected_checksum) {
                i++;
                continue;
            }
            if (len == 0) {
                out_payload = NULL;
                *out_type = type;
                return 0;
            }
            if (i + 4 + len <= stream_len) {
                out_payload = (uint8_t *)malloc(len);
                if (!out_payload) return -1;
                for (int j = 0; j < len; ++j) {
                    out_payload[j] = stream[i + 1 + j];
                }
                *out_type = type;
                return len;
            }
            i++;
        } else {
            i++;
        }
    }
    return -1;
}
