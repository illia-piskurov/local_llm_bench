class InMemoryKVStore:
    def __init__(self):
        self.data = {}
        self.transactions = []
        self.watches = {}
        self.snapshots = {}

    def set(self, key, value):
        current_value = self.get(key)
        if current_value != value:
            self.data[key] = value
            if key in self.watches:
                print(f"WATCH {key} {current_value} -> {value}")

    def get(self, key):
        return self.data.get(key, "NULL")

    def delete(self, key):
        if key in self.data:
            current_value = self.data[key]
            del self.data[key]
            if key in self.watches:
                print(f"WATCH {key} {current_value} -> NULL")
        else:
            raise KeyError("Key not found for deletion")

    def begin_transaction(self):
        self.transactions.append({})

    def commit(self):
        if not self.transactions:
            return "NO TRANSACTION"
        transaction = self.transactions.pop()
        parent = self.transactions[-1] if self.transactions else None
        for k, v in transaction.items():
            parent[k] = v

    def rollback(self):
        if not self.transactions:
            return "NO TRANSACTION"
        self.transactions.pop()

    def count(self, value):
        current_data = self.data.copy()
        for trans in reversed(self.transactions):
            for k, v in trans.items():
                current_data[k] = v
        count = sum(1 for val in current_data.values() if val == value)
        return str(count)

    def watch_notifications(self):
        notifications = []
        current_data = self.data.copy()
        for trans in reversed(self.transactions):
            for k, v in trans.items():
                current_data[k] = v
        for key, old_val in self.watches.items():
            new_val = current_data.get(key, "NULL")
            if old_val != new_val:
                notifications.append(f"WATCH {key} {old_val} -> {new_val}")
        return "\n".join(notifications)

    def snapshot(self, name):
        state = {
            'data': self.data.copy(),
            'transactions': [t.copy() for t in self.transactions],
            'watches': self.watches.copy(),
            'snapshots': self.snapshots.copy()
        }
        self.snapshots[name] = state

    def restore(self, name):
        if name not in self.snapshots:
            raise ValueError("Unknown snapshot")
        state = self.snapshots[name]
        self.data = state['data'].copy()
        for i in range(len(self.transactions)):
            self.transactions[i] = state['transactions'][i].copy()
        self.watches = state['watches'].copy()
        self.snapshots = state['snapshots'].copy()

def run(program: str) -> list[str]:
    store = InMemoryKVStore()
    output = []
    lines = (line.strip() for line in program.splitlines() if line.strip())
    for line in lines:
        parts = line.split()
        cmd = parts[0]
        if cmd == "SET":
            key, value = parts[1], " ".join(parts[2:])
            store.set(key, value)
        elif cmd == "GET":
            key = parts[1]
            output.append(store.get(key))
        elif cmd == "DELETE":
            key = parts[1]
            try:
                store.delete(key)
            except KeyError:
                pass
        elif cmd == "BEGIN":
            store.begin_transaction()
        elif cmd == "COMMIT":
            output.append(store.commit())
        elif cmd == "ROLLBACK":
            output.append(store.rollback())
        elif cmd == "COUNT":
            value = parts[1]
            output.append(store.count(value))
        elif cmd == "WATCH":
            key = parts[1]
            store.watches[key] = store.get(key)
        elif cmd == "SNAPSHOT":
            name = parts[1]
            store.snapshot(name)
        elif cmd == "RESTORE":
            name = parts[1]
            store.restore(name)
    print(store.watch_notifications())
    return output
