local BT = {}

-- Execution statuses (unchanged)
BT.SUCCESS = "SUCCESS"
BT.FAILURE = "FAILURE"
BT.RUNNING = "RUNNING"

-- Blackboard implementation (unchanged)
BT.Blackboard = {}
function BT.Blackboard.new()
    local self = setmetatable({}, { __index = BT.Blackboard })
    self._data = {}
    self._watchers = {}
    return self
end

function BT.Blackboard:get(self, key, default)
    return self._data[key] or default
end

function BT.Blackboard:set(self, key, value)
    local old_value = self._data[key]
    if old_value ~= value then
        self._data[key] = value
        for _, callback in ipairs(self._watchers[key] or {}) do
            callback(key, value, old_value)
        end
    end
end

function BT.Blackboard:watch(self, key, callback)
    if not self._watchers[key] then
        self._watchers[key] = {}
    end
    table.insert(self._watchers[key], callback)
end

-- Decorators (unchanged)
function BT.Inverter(child)
    local node = new_node()
    function node:tick(ctx)
        local result = child:tick(ctx)
        if result == BT.SUCCESS then return BT.FAILURE end
        if result == BT.FAILURE then return BT.SUCCESS end
        return result
    end

    function node:reset()
        child:reset()
    end
    return setmetatable(node, { __index = BT })
end

function BT.Cooldown(child, ticks)
    local node = new_node()
    local cooldown = 0

    function node:tick(ctx)
        if cooldown > 0 then
            cooldown = max(0, cooldown - 1)
            return BT.FAILURE
        end

        local result = child:tick(ctx)
        if result == BT.SUCCESS then
            cooldown = ticks
            return BT.RUNNING
        else
            return result
        end
    end

    function node:reset()
        cooldown = 0
        child:reset()
    end
    return setmetatable(node, { __index = BT })
end

-- Base Node class (unchanged)
local function new_node()
    return setmetatable({}, {
        __index = function(self, key)
            error("Attempt to access non-existent method: " .. key)
        end,
        tick = function() end,
        reset = function() end
    })
end

-- Action Node (unchanged)
function BT.Action(fn)
    local node = new_node()

    function node:tick(ctx)
        local result = fn(ctx)
        if result == true then return BT.SUCCESS end
        if result == false then return BT.FAILURE end
        if result == BT.SUCCESS then return BT.SUCCESS end
        if result == BT.FAILURE then return BT.FAILURE end
        return BT.RUNNING
    end

    function node:reset()
        -- No state to reset for Action nodes
    end

    return setmetatable(node, { __index = BT })
end

-- Condition Node (unchanged)
function BT.Condition(predicate)
    local node = new_node()

    function node:tick(ctx)
        if predicate(ctx) then return BT.SUCCESS else return BT.FAILURE end
    end

    function node:reset()
        -- No state to reset for Condition nodes
    end

    return setmetatable(node, { __index = BT })
end

-- Sequence Node (AND) - unchanged with cascading reset
function BT.Sequence(children)
    local node = new_node()
    children = children or {}
    local current_index = 1

    function node:tick(ctx)
        if current_index > #children then return BT.SUCCESS end
        local child = children[current_index]
        local result = child:tick(ctx)

        if result == BT.FAILURE then return BT.FAILURE end
        if result == BT.RUNNING then
            current_index = current_index + 1
            return BT.RUNNING
        else
            current_index = current_index + 1
            return BT.SUCCESS
        end
    end

    function node:reset()
        for _, child in ipairs(children) do child:reset() end
        current_index = 1
    end

    return setmetatable(node, { __index = BT })
end

-- Selector Node (OR) - unchanged with cascading reset
function BT.Selector(children)
    local node = new_node()
    children = children or {}
    local current_index = 1

    function node:tick(ctx)
        if current_index > #children then return BT.FAILURE end
        local child = children[current_index]
        local result = child:tick(ctx)

        if result == BT.SUCCESS then return BT.SUCCESS end
        if result == BT.RUNNING then
            current_index = current_index + 1
            return BT.RUNNING
        else
            current_index = current_index + 1
            return BT.FAILURE
        end
    end

    function node:reset()
        for _, child in ipairs(children) do child:reset() end
        current_index = 1
    end

    return setmetatable(node, { __index = BT })
end

-- Async Action Node with coroutine support
function BT.AsyncAction(coroutine_fn)
    local node = new_node()
    local co = nil
    local wait_ticks = 0
    local ctx = {}

    function node:tick(ctx)
        if not co then
            co = coroutine.create(function()
                local result = coroutine_fn(ctx)
                if type(result) == "table" and result.wait_ticks then
                    wait_ticks = result.wait_ticks
                    return BT.RUNNING
                else
                    return result
                end
            end)
        end

        if wait_ticks > 0 then
            wait_ticks = max(0, wait_ticks - 1)
            return BT.RUNNING
        end

        local status, result = coroutine.resume(co or {}, ctx)
        if status == false then co = nil end

        if result == BT.SUCCESS then return BT.SUCCESS end
        if result == BT.FAILURE then return BT.FAILURE end
        return BT.RUNNING
    end

    function node:reset()
        co = nil
        wait_ticks = 0
    end

    return setmetatable(node, { __index = BT })
end
