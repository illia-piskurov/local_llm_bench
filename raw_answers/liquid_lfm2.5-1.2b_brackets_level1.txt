def is_balanced(s: str) -> bool:
    stack = []
    opening_scopes = {'(', '{', '['}
    closing_scopes = {')': '(', '}': '{', ']': '['}
    
    for char in s:
        if char in opening_scopes:
            stack.append(char)
        elif char in closing_scopes:
            if not stack or stack.pop() != closing_scopes[char]:
                return False
    return not stack
