from heapq import heappush, heappop

def critical_path(tasks: Dict[str, Tuple[int, List[str], int]]) -> Optional[int]:
    def find_longest_path(task, visited, path_length):
        visited.add(task)
        longest = path_length
        for dep in tasks[task][1]:
            if dep not in visited:
                longest = max(longest, find_longest_path(dep, visited, path_length + tasks[dep][0]))
        visited.remove(task)
        return longest

    visited = set()
    longest_path = 0
    for task in tasks:
        if task not in visited:
            path_length = find_longest_path(task, visited, 0)
            longest_path = max(longest_path, path_length)
    return None if longest_path == 0 else longest_path
