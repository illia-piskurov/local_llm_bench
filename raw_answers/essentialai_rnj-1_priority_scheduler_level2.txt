def critical_path(tasks: dict[str, tuple[int, list[str], int]]) -> int | None:
    from collections import defaultdict

    def has_cycle():
        visited = set()
        rec_stack = set()

        def dfs(task):
            if task in rec_stack:
                return True
            if task in visited:
                return False
            visited.add(task)
            rec_stack.add(task)
            for dep in tasks[task][1]:
                if dfs(dep):
                    return True
            rec_stack.remove(task)
            return False

        for task in tasks:
            if dfs(task):
                return None
        return True

    def topological_sort():
        indegree = {task: 0 for task in tasks}
        graph = defaultdict(list)

        for task, (_, deps, _) in tasks.items():
            for dep in deps:
                graph[dep].append(task)
                indegree[task] += 1

        queue = deque([task for task in tasks if indegree[task] == 0])
        sorted_tasks = []

        while queue:
            current_task = queue.popleft()
            sorted_tasks.append(current_task)
            for neighbor in graph[current_task]:
                indegree[neighbor] -= 1
                if indegree[neighbor] == 0:
                    queue.append(neighbor)

        return sorted_tasks

    def calculate_path_length():
        task_times = defaultdict(int)
        max_time = 0

        for task in topological_sort():
            duration, deps, _ = tasks[task]
            start_time = max((task_times[dep] for dep in deps), default=0)
            end_time = start_time + duration
            task_times[task] = end_time
            max_time = max(max_time, end_time)

        return max_time

    if not has_cycle():
        return calculate_path_length()
    else:
        return None
