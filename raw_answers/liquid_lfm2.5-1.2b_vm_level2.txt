def run(program: str) -> list[str]:
    stack = []
    output = []
    lines = program.split('\n')
    for i, line in enumerate(lines):
        if line.startswith('#'):
            continue
        parts = line.split()
        cmd = parts[0]
        if cmd == 'PUSH':
            try:
                stack.append(int(parts[1]))
            except ValueError:
                print(f"Ошибка: строка {i+1} содержит некорректное число для PUSH")
        elif cmd == 'POP':
            if len(stack) < 1:
                print(f"Ошибка: стек пуст при POP (строка {i+1})")
            else:
                output.append(str(stack.pop()))
        elif cmd == 'ADD':
            if len(stack) < 2:
                print(f"Ошибка: стек недостаточно для ADD (строка {i+1})")
            else:
                output.append(str(stack.pop() + stack.pop()))
        elif cmd == 'SUB':
            if len(stack) < 2:
                print(f"Ошибка: стек недостаточно для SUB (строка {i+1})")
            else:
                output.append(str(stack.pop() - stack.pop()))
        elif cmd == 'MUL':
            if len(stack) < 2:
                print(f"Ошибка: стек недостаточно для MUL (строка {i+1})")
            else:
                output.append(str(stack.pop() * stack.pop()))
        elif cmd == 'DIV':
            if len(stack) < 2:
                print(f"Ошибка: стек недостаточно для DIV (строка {i+1})")
                continue
            try:
                b = stack.pop()
                a = stack.pop()
                result = a / b
                if result == int(result):
                    output.append(str(int(result)))
                else:
                    output.append(str(result))
            except ZeroDivisionError:
                print(f"Ошибка: деление на ноль при DIV (строка {i+1})")
        elif cmd == 'DUP':
            if len(stack) < 1:
                print(f"Ошибка: стек пуст при DUP (строка {i+1})")
            else:
                stack.append(stack[-1])
        elif cmd == 'SWAP':
            if len(stack) < 2:
                print(f"Ошибка: стек недостаточно для SWAP (строка {i+1})")
            else:
                stack[0], stack[1] = stack[1], stack[0]
        elif cmd == 'PRINT':
            if len(stack) < 1:
                print(f"Ошибка: стек пуст при PRINT (строка {i+1})")
            else:
                output.append(str(stack[-1]))
        elif cmd == 'LABEL':
            L = line.find('LABEL')
            if L != -1:
                name = lines[L + 1]
                print(f"Метка: {name}")
                continue
        elif cmd == 'JMP':
            if i < len(lines):
                JMP = lines[i + 1]
                print(f"Переход к метке: {JMP}")
                continue
        elif cmd == 'JZ':
            if int(parts[1]) == 0:
                if len(stack) >= 2:
                    stack.pop()
                else:
                    print(f"Ошибка: стек пуст при JZ (строка {i+1})")
            else:
                print(f"Ошибка: неправильное условие JZ в строке {i+1}")
        elif cmd == 'JNZ':
            if int(parts[1]) != 0:
                if len(stack) >= 2:
                    stack.pop()
                else:
                    print(f"Ошибка: стек недостаточно при JNZ (строка {i+1})")
        else:
            print(f"Неверный команда в строке {i+1}")
    return output
