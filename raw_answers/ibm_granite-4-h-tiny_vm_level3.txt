def run(program: str) -> list[str]:
    lines = [line.strip() for line in program.splitlines() if line.strip() and not line.strip().startswith("#")]
    stack = []
    call_stack = []
    memory = {}
    output = []
    label_map = {}
    line_number = 0

    def error(msg):
        raise ValueError(f"Error at line {line_number}: {msg}")

    current_label = None
    for i, line in enumerate(lines):
        line_number += 1
        parts = line.split()
        cmd = parts[0]

        try:
            if cmd == "LABEL":
                current_label = parts[1]
            elif cmd == "JMP":
                target = label_map[parts[1]]
                stack.pop()  # value is discarded but must be popped
                line_number = target - 1
            elif cmd == "JZ":
                if len(stack) < 1:
                    error("not enough values on the stack for JZ")
                a = stack.pop()
                if a == 0:
                    target = label_map[parts[1]]
                    line_number = target - 1
            elif cmd == "JNZ":
                if len(stack) < 1:
                    error("not enough values on the stack for JNZ")
                a = stack.pop()
                if a != 0:
                    target = label_map[parts[1]]
                    line_number = target - 1
            elif cmd == "PUSH":
                value = int(parts[1])
                stack.append(value)
            elif cmd == "POP":
                if not stack:
                    error("stack underflow")
                stack.pop()
            elif cmd == "ADD":
                if len(stack) < 2:
                    error("not enough values on the stack for ADD")
                a, b = stack.pop(), stack.pop()
                stack.append(a + b)
            elif cmd == "SUB":
                if len(stack) < 2:
                    error("not enough values on the stack for SUB")
                a, b = stack.pop(), stack.pop()
                stack.append(b - a)
            elif cmd == "MUL":
                if len(stack) < 2:
                    error("not enough values on the stack for MUL")
                a, b = stack.pop(), stack.pop()
                stack.append(a * b)
            elif cmd == "DIV":
                if len(stack) < 2:
                    error("not enough values on the stack for DIV")
                a, b = stack.pop(), stack.pop()
                if b == 0:
                    error("division by zero")
                stack.append(b // a)
            elif cmd == "DUP":
                if not stack:
                    error("stack underflow")
                stack.append(stack[-1])
            elif cmd == "SWAP":
                if len(stack) < 2:
                    error("not enough values on the stack for SWAP")
                stack[-1], stack[-2] = stack[-2], stack[-1]
            elif cmd == "PRINT":
                if not stack:
                    error("stack underflow")
                output.append(str(stack[-1]))
            elif cmd == "CALL":
                target = label_map[parts[1]]
                call_stack.append(line_number + 1)
                line_number = target - 1
            elif cmd == "RET":
                if not call_stack:
                    error("no return address available")
                line_number = call_stack.pop()
            elif cmd == "EQ":
                if len(stack) < 2:
                    error("not enough values on the stack for EQ")
                a, b = stack.pop(), stack.pop()
                stack.append(1 if a == b else 0)
            elif cmd == "GT":
                if len(stack) < 2:
                    error("not enough values on the stack for GT")
                a, b = stack.pop(), stack.pop()
                stack.append(1 if a > b else 0)
            elif cmd == "LT":
                if len(stack) < 2:
                    error("not enough values on the stack for LT")
                a, b = stack.pop(), stack.pop()
                stack.append(1 if a < b else 0)
            elif cmd.startswith("STORE"):
                var = parts[1]
                value = stack.pop()
                memory[var] = value
            elif cmd == "LOAD":
                var = parts[1]
                if var not in memory:
                    error(f"variable {var} has not been stored")
                stack.append(memory[var])
            else:
                error(f"unknown command: {cmd}")
        except (IndexError, ValueError) as e:
            raise ValueError(f"Syntax or value error at line {line_number}: {e}")

    return output

                if not call_stack:
                    error("no return address available")
                line_number = call_stack.pop()
            elif cmd == "EQ":
                if len(stack) < 2:
                    error("not enough values on the stack for EQ")
                a, b = stack.pop(), stack.pop()
                stack.append(1 if a == b else 0)
            elif cmd == "GT":
                if len(stack) < 2:
                    error("not enough values on the stack for GT")
                a, b = stack.pop(), stack.pop()
                stack.append(1 if a > b else 0)
            elif cmd == "LT":
                if len(stack) < 2:
                    error("not enough values on the stack for LT")
                a, b = stack.pop(), stack.pop()
                stack.append(1 if a < b else 0)
            elif cmd.startswith("STORE"):
                var = parts[1]
                value = stack.pop()
                memory[var] = value
            elif cmd == "LOAD":
                var = parts[1]
                if var not in memory:
                    error(f"variable {var} has not been stored")
                stack.append(memory[var])
            else:
                error(f"unknown command: {cmd}")
        except (IndexError, ValueError) as e:
            raise ValueError(f"Syntax or value error at line {line_number}: {e}")

    return output

    output = []

                    line_number = target - 1

                a, b = stack.pop(), stack.pop()
                stack.append(a + b)

            elif cmd == "LT":
                if len(stack) < 2:
                    error("not enough values on the stack for LT")
                a, b = stack.pop(), stack.pop()
                stack.append(1 if a < b else 0)
            elif cmd.startswith("STORE"):
