import datetime
from typing import List, Tuple, Set

# --- Helper Functions for Cron Parsing ---

def parse_range(spec: str, min_val: int, max_val: int) -> Set[int]:
    """Parses a cron field specification (e.g., '1-5', '1,5') into a set of valid values."""
    values = set()
    parts = spec.split(',')
    for part in parts:
        part = part.strip()
        if '-' in part:
            try:
                start_str, end_str = map(int, part.split('-'))
                # Ensure range is within bounds and ordered correctly
                start = max(min_val, start_str)
                end = min(max_val, end_str)
                if start <= end:
                    values.update(range(start, end + 1))
            except ValueError:
                # Handle malformed ranges gracefully (though input validation assumes correctness)
                pass
        else:
            try:
                val = int(part)
                if min_val <= val <= max_val:
                    values.add(val)
            except ValueError:
                pass
    return values

def parse_cron_field(spec: str, min_val: int, max_val: int) -> Set[int]:
    """Parses a single cron field specification."""
    if spec == '*':
        return set(range(min_val, max_val + 1))
    return parse_range(spec, min_val, max_val)

def parse_cron_expression(cron_str: str) -> Tuple[Set[int], Set[int], Set[int], Set[int], Set[int]]:
    """
    Parses the 5-field cron string into sets of allowed values.
    Returns: (Minute_set, Hour_set, DoM_set, Month_set, DoW_set)
    """
    try:
        fields = cron_str.split()
        if len(fields) != 5:
            raise ValueError("Cron expression must contain exactly 5 fields.")

        # Field ranges:
        # Minute (0-59)
        minute_set = parse_cron_field(fields[0], 0, 59)
        # Hour (0-23)
        hour_set = parse_cron_field(fields[1], 0, 23)
        # Day of Month (1-31)
        dom_set = parse_cron_field(fields[2], 1, 31)
        # Month (1-12)
        month_set = parse_cron_field(fields[3], 1, 12)
        # Day of Week (0-6)
        dow_set = parse_cron_field(fields[4], 0, 6)

        return minute_set, hour_set, dom_set, month_set, dow_set

    except Exception as e:
        raise ValueError(f"Invalid cron expression format or range: {e}")


# --- Core Scheduler Logic ---

def matches_cron(dt: datetime.datetime, parsed_fields) -> bool:
    """
    Checks if a given datetime object matches the parsed cron constraints.

    dt must be timezone-aware (UTC).
    parsed_fields = (M_set, H_set, DoM_set, Mo_set, DoW_set)
    """
    minute_set, hour_set, dom_set, month_set, dow_set = parsed_fields

    # 1. Check Minute
    if dt.minute not in minute_set:
        return False

    # 2. Check Hour
    if dt.hour not in hour_set:
        return False

    # 3. Check Day of Month (DoM)
    if dt.day not in dom_set:
        return False

    # 4. Check Month (assuming month is 1-indexed)
    if dt.month not in month_set:
        return False

    # 5. Check Day of Week (DoW)
    # weekday() returns 0 (Monday) to 6 (Sunday). Cron uses 0 (Sunday) to 6 (Saturday).
    # We need to convert Python's weekday() result:
    # Monday (0) -> 1, Sunday (6) -> 0.
    # Cron DoW: 0=Sun, ..., 6=Sat.
    # Python weekday: 0=Mon, ..., 6=Sun.

    # If dt is Sunday (weekday() == 6), Cron DoW must be 0.
    # If dt is Monday (weekday() == 0), Cron DoW must be 1.
    # The required mapping is: target_dow = (dt.weekday() + 1) % 7
    # Wait, the requirement is: 0=Sun, ..., 6=Sat.
    # Python's dt.weekday(): Mon=0, Tue=1, ..., Sun=6.
    # If dt is Sunday (dt.weekday() == 6), Cron DoW must be 0.
    # If dt is Monday (dt.weekday() == 0), Cron DoW must be 1.
    # The correct mapping is: target_dow = (dt.weekday() + 1) % 7
    # Example: Sunday (6) -> (6+1)%7 = 0. Monday (0) -> (0+1)%7 = 1. This works.
    cron_dow = (dt.weekday() + 1) % 7

    if cron_dow not in dow_set:
        return False

    return True


def next_runs(cron: str, from_time: str, count: int) -> List[str]:
    """
    Calculates the next 'count' run times strictly after from_time.

    :param cron: The 5-field cron expression string.
    :param from_time: ISO 8601 UTC starting time string (e.g., "2026-08-18T10:00:00.000Z").
    :param count: The number of future runs to find.
    :return: List of ISO 8601 UTC strings for the next runs.
    """
    # --- Step 1: Parse Input and Setup ---
    try:
        minute_set, hour_set, dom_set, month_set, dow_set = parse_cron_expression(cron)
    except ValueError as e:
        raise ValueError(f"Error parsing cron expression: {e}")

    # Parse starting time string into timezone-aware datetime object
    try:
        # Handle potential 'Z' suffix and parse to UTC
        if from_time.endswith('Z'):
            from_time_dt = datetime.datetime.strptime(from_time[:-1], "%Y-%m-%dT%H:%M:%S.%f").replace(tzinfo=datetime.timezone.utc)
        else:
            from_time_dt = datetime.datetime.strptime(from_time, "%Y-%m-%dT%H:%M:%S.%f").replace(tzinfo=datetime.timezone.utc)
    except ValueError:
        raise ValueError("Invalid from_time format. Must be ISO 8601 UTC.")

    # We must start searching from the next second after from_time
    current_dt = from_time_dt + datetime.timedelta(seconds=1)

    # --- Step 2: Search for Runs ---
    found_runs = []
    search_dt = current_dt

    while len(found_runs) < count:
        # Check if the current time matches the cron expression
        if matches_cron(search_dt, (minute_set, hour_set, dom_set, month_set, dow_set)):
            # Found a match. Store it and increment the count.
            found_runs.append(search_dt)

        # Move to the next second for the search
        search_dt += datetime.timedelta(seconds=1)

    # --- Step 3: Format Output ---
    # Convert found datetime objects back to the required ISO 8601 UTC string format.
    # We truncate microseconds/milliseconds to match the required output precision (00:00.000Z)
    output_list = []
    for dt in found_runs:
        # Format to YYYY-MM-DDTHH:MM:SS.000Z
        # We use strftime for clean formatting and append 'Z'
        iso_str = dt.strftime("%Y-%m-%dT%H:%M:%S") + ".000Z"
        output_list.append(iso_str)

    return output_list
