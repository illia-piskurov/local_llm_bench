def topo_sort(tasks: dict[str, list[str]]) -> list[str] | None:
    from collections import deque

    in_degree = {task: 0 for task in tasks}
    for deps in tasks.values():
        for dep in deps:
            in_degree[dep] += 1

    queue = deque([task for task in in_degree if in_degree[task] == 0])
    sorted_order = []

    while queue:
        task = queue.popleft()
        sorted_order.append(task)

        for dep in tasks[task]:
            in_degree[dep] -= 1
            if in_degree[dep] == 0:
                queue.append(dep)

    if len(sorted_order) != len(tasks):
        return None

    return sorted_order

def critical_path(tasks: dict[str, tuple[int, list[str]]]) -> int | None:
    topo_sorted_tasks = topo_sort({task: deps for task, deps in tasks.items()})
    if topo_sorted_tasks is None:
        return None

    durations = {task: duration for task, (duration, _) in tasks.items()}
    total_duration = 0

    for task in topo_sorted_tasks:
        duration, _ = tasks[task]
        total_duration += duration

    return total_duration
