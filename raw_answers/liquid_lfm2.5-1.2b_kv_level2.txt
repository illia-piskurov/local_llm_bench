import json
from typing import List

class InMemoryKeyValueStore:
    def __init__(self):
        self.store = {}
        self.trx = None
        self.trx_active = False
        self.watch_keys = set()

    def begin(self):
        if not self.trx_active:
            self.trx = {}
            self.trx_active = True

    def commit(self):
        if self.trx_active:
            self.store = self.trx.copy()
            self.trx_active = False
            return "NO TRANSACTION"
        else:
            return "NO TRANSACTION"

    def rollback(self):
        if self.trx_active:
            self.store = {}
            self.trx_active = False
            return "NO TRANSACTION"
        else:
            return "NO TRANSACTION"

    def set(self, key: str, value: str):
        if self.trx_active:
            old_value = self.get(key)
            if old_value == value:
                return True
            self.watch_keys.add(key)
            self.trx[key] = value
            return False
        else:
            self.store[key] = value
            return False

    def get(self, key: str) -> str:
        if self.trx_active:
            if key in self.trx:
                return self.trx[key]
            else:
                return "NULL"
        else:
            return self.store.get(key, "NULL")

    def delete(self, key: str):
        if self.trx_active:
            if key in self.trx:
                del self.trx[key]
                self.watch_keys.discard(key)
                return True
            else:
                return False
        else:
            self.store.pop(key, None)
            self.watch_keys.discard(key)
            return True

    def run(self, program: str) -> List[str]:
        result = []
        if program == "BEGIN":
            self.begin()
        elif program == "GET":
            result.append(self.get(program.split()[1]))
        elif program == "COMMIT":
            result.append(self.commit())
        elif program == "ROLLBACK":
            result.append(self.rollback())
        elif program == "COUNT":
            if self.watch_keys:
                count = len(self.store)
                old_value = self.get("count")
                if count == int(old_value):
                    result.append(f"COUNT {int(old_value)}")
                else:
                    result.append(f"COUNT {count}")
            else:
                result.append("NO TRANSACTION")
        elif program == "WATCH":
            if self.watch_keys:
                key = next(iter(self.watch_keys))
                old_value = self.get(key)
                if old_value == "NULL":
                    result.append(f"WATCH {key} NULL -> {value}")
                else:
                    result.append(f"WATCH {key} {old_value} -> {value}")
            else:
                result.append("NO TRANSACTION")
        return result
