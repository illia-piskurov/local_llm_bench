    <style>
        :root {

        body {

<body>

    <h1>SNAKE</h1>
    <div id="score">Счёт: 0</div>
    
    <div id="game-container">
        <canvas id="gameCanvas" width="400" height="400"></canvas>
        <div id="game-over">
            <h2>GAME OVER</h2>
            <p id="final-score"></p>
            <p>Нажмите <b style="color: white;">SPACE</b> или <b style="color: white;">R</b> для рестарта</p>
        </div>
    </div>

    <div class="controls-hint">УПРАВЛЕНИЕ: СТРЕЛКИ / WASD</div>

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreElement = document.getElementById('score');
        const gameOverScreen = document.getElementById('game-over');
        const finalScoreElement = document.getElementById('final-score');
        const container = document.getElementById('game-container');

        const gridSize = 20;
        const tileCount = canvas.width / gridSize;

        let snake, food, dx, dy, score, gameRunning, gameSpeed, lastUpdateTime;
        let particles = [];

        function initGame() {
            snake = [
                {x: 10, y: 10},
                {x: 9, y: 10},
                {x: 8, y: 10}
            ];
            
            dx = 1;
            dy = 0;
            score = 0;
            gameRunning = true;
            gameSpeed = 100; 
            lastUpdateTime = 0;

            scoreElement.innerText = `Счёт: ${score}`;
            gameOverScreen.classList.remove('show');
            container.classList.remove('shake');
            
            createFood();
            requestAnimationFrame(gameLoop);
        }

        function createFood() {
            food = {
                x: Math.floor(Math.random() * tileCount),
                y: Math.floor(Math.random() * tileCount)
            };
            // Проверка, чтобы еда не появилась внутри змейки
            snake.forEach(part => {
                if (part.x === food.x && part.y === food.y) {
                    createFood();
                }
            });
        }

        // Система частиц
        class Particle {
            constructor(x, y, color) {
                this.x = x;
                this.y = y;
                this.color = color;
                this.size = Math.random() * 4 + 1;
                this.speedX = (Math.random() - 0.5) * 8;
                this.speedY = (Math.random() - 0.5) * 8;
                this.life = 1.0;
                this.decay = Math.random() * 0.02 + 0.02;
            }

            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                this.life -= this.decay;
            }

            draw(ctx) {
                ctx.globalAlpha = this.life;
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.globalAlpha = 1.0;
            }
        }

        function gameLoop(currentTime) {
            if (!gameRunning) return;

            const deltaTime = currentTime - lastUpdateTime;

            if (deltaTime > gameSpeed) {
                updateLogic();
                lastUpdateTime = currentTime;
            }

            draw(currentTime);
            requestAnimationFrame(gameLoop);
        }

        function updateLogic() {
            const head = { x: snake[0].x + dx, y: snake[0].y + dy };

            // Столкновение со стенами
            if (head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {
                return gameOver();
            }

            // Столкновение с самим собой
            for (let i = 0; i < snake.length; i++) {
                if (snake[i].x === head.x && snake[i].y === head.y) {
                    return gameOver();
                }
            }

            snake.unshift(head);

            // Поедание еды
            if (head.x === food.x && head.y === food.y) {
                score += 10;
                scoreElement.innerText = `Счёт: ${score}`;
                
                // Создаем частицы при поедании
                for(let i = 0; i < 15; i++) {
                    particles.push(new Particle(food.x * gridSize + gridSize/2, food.y * gridSize + gridSize/2, '#ff0055'));
                }

                createFood();
                if (score % 50 === 0 && gameSpeed > 50) gameSpeed -= 5;
            } else {
                snake.pop();
            }
        }

        function draw(currentTime) {
            // Очистка фона с легким шлейфом
            ctx.fillStyle = 'rgba(5, 5, 10, 0.3)';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Рисуем еду с пульсацией
            const pulse = Math.sin(currentTime / 100) * 3;
            ctx.shadowBlur = 15;
            ctx.shadowColor = '#ff0055';
            ctx.fillStyle = '#ff0055';
            ctx.beginPath();
            ctx.arc(
                food.x * gridSize + gridSize/2, 
                food.y * gridSize + gridSize/2, 
                (gridSize/2 - 2) + pulse/2, 
                0, Math.PI * 2
            );
            ctx.fill();
            ctx.shadowBlur = 0;

            // Рисуем змейку
            snake.forEach((part, index) => {
                const isHead = index === 0;
                const gradient = ctx.createLinearGradient(
                    part.x * gridSize, part.y * gridSize, 
                    (part.x + 1) * gridSize, (part.y + 1) * gridSize
                );
                
                if (isHead) {
                    gradient.addColorStop(0, '#00ff9d');
                    gradient.addColorStop(1, '#00cc7a');
                    ctx.shadowBlur = 20;
                    ctx.shadowColor = '#00ff9d';
                } else {
                    gradient.addColorStop(0, '#00cc7a');
                    gradient.addColorStop(1, '#00885a');
                    ctx.shadowBlur = 5;
                    ctx.shadowColor = '#00cc7a';
                }

                ctx.fillStyle = gradient;
                
                // Рисуем сегменты как скругленные прямоугольники
                ctx.beginPath();
                ctx.roundRect(
                    part.x * gridSize + 1, 
                    part.y * gridSize + 1, 
                    gridSize - 2, 
                    gridSize - 2, 
                    isHead ? 8 : 4
                );
                ctx.fill();
            });
            ctx.shadowBlur = 0;

            // Рисуем частицы
            particles = particles.filter(p => p.life > 0);
            particles.forEach(p => {
                p.update();
                p.draw(ctx);
            });
        }

        function gameOver() {
            gameRunning = false;
            finalScoreElement.innerText = `Ваш результат: ${score}`;
            gameOverScreen.classList.add('show');
            container.classList.add('shake');
        }

        function handleInput(e) {
            const key = e.key.toLowerCase();
            
            if ((key === 'arrowup' || key === 'w') && dy === 0) {
                dx = 0;
                dy = -1;
            }
            if ((key === 'arrowdown' || key === 's') && dy === 0) {
                dx = 0;
                dy = 1;
            }
            if ((key === 'arrowleft' || key === 'a') && dx === 0) {
                dx = -1;
                dy = 0;
            }
            if ((key === 'arrowright' || key === 'd') && dx === 0) {
                dx = 1;
                dy = 0;
            }

            if ((key === ' ' || key === 'r') && !gameRunning) {
                initGame();
            }
        }

        window.addEventListener('keydown', handleInput);

        // Запуск
        initGame();
    </script>
</body>
</html>
