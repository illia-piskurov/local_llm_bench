```python
def max_depth(s: str) -> int:
    """
    Вычисляет максимальную глубину вложенности скобок при сканировании строки слева направо.
    """
    max_depth = 0
    current_depth = 0
    for char in s:
        if char == '(':
            current_depth += 1
            max_depth = max(max_depth, current_depth)
        elif char == ')':
            current_depth -= 1
            if current_depth < 0:
                return -1  # Неверный порядок скобок
        else:
            pass

    return max_depth
```
