def compile_query(query: dict) -> dict:
    """
    Компилирует AST-дерево запроса в параметризованный SQL.
    
    Args:
        query: Словарь с описанием запроса (см. входной словарь).
        
    Returns:
        {"sql": str, "params": list}
    """
    sql = []
    params = []

    # --- SELECT ---
    if 'select' in query and query['select']:
        columns = ', '.join(query['select'])
        sql.append(f'SELECT {columns}')
    else:
        sql.append('SELECT *')

    # --- FROM ---
    table = query.get('table', '')
    if table:
        sql.append(f'FROM {table}')

    # --- WHERE ---
    where_clause = ''
    if 'where' in query and query['where']:
        where_conditions, params = _compile_where(query['where'], params)
        if where_conditions:
            where_clause = f'WHERE {where_conditions}'
            sql.append(where_clause)

    # --- JOIN (опционально, для полноты) ---
    joins = query.get('join', [])
    for join in joins:
        table_name = join['table']
        on_condition, params = _compile_where(join['on'], params)
        if on_condition:
            sql.append(f'JOIN {table_name} ON {on_condition}')

    # --- GROUP BY (опционально) ---
    group_by = query.get('groupBy', [])
    if group_by:
        fields = ', '.join(group_by)
        sql.append(f'GROUP BY {fields}')

    # --- HAVING (опционально) ---
    having_clause = ''
    if 'having' in query and query['having']:
        having_conditions, params = _compile_where(query['having'], params)
        if having_conditions:
            having_clause = f'HAVING {having_conditions}'
            sql.append(having_clause)

    # --- ORDER BY ---
    order_by = query.get('orderBy', [])
    if order_by:
        order_parts = []
        for item in order_by:
            field = item['field']
            direction = item.get('dir', 'ASC')
            order_parts.append(f'{field} {direction}')
        sql.append(f'ORDER BY {", ".join(order_parts)}')

    # --- LIMIT / OFFSET ---
    limit = query.get('limit')
    offset = query.get('offset')
    if limit is not None:
        sql.append(f'LIMIT {limit}')
    if offset is not None:
        sql.append(f'OFFSET {offset}')

    return {'sql': ' '.join(sql), 'params': params}


def _compile_where(conditions: list, params: list) -> tuple:
    """Рекурсивно компилирует дерево условий WHERE."""
    result = []
    
    for cond in conditions:
        if isinstance(cond, dict):
            # Логический оператор (AND/OR)
            op = cond.get('op', '')
            left, right = _compile_condition(cond['left'], params)
            
            if op == 'AND':
                result.append(f'({left} AND {right})')
            elif op == 'OR':
                result.append(f'({left} OR {right})')
        else:
            # Простое условие: { field: str, op: str, value: any }
            field = cond['field']
            operator = cond['op']
            value = cond['value']
            
            if isinstance(value, (int, float)):
                params.append(value)
                result.append(f'{field} {operator} $1')
            elif isinstance(value, str):
                params.append(value)
                result.append(f'{field} {operator} $1')
            else:
                params.append(value)
                result.append(f'{field} {operator} $1')

    return ' '.join(result), params


def _compile_condition(cond: dict, params: list) -> str:
    """Компилирует одно условие (левая/правая часть)."""
    field = cond['field']
    operator = cond['op']
    
    value = cond.get('value')
    if isinstance(value, (int, float)):
        return f'{field} {operator} $1'
    elif isinstance(value, str):
        return f'{field} {operator} $1'
    else:
        return f'{field} {operator} $1'

if __name__ == '__main__':

    query1 = {

**Ключевые особенности:**
- **Параметризация**: все значения подставляются через `$1`, `$2`... (подходит для PostgreSQL, MySQL)
- **Рекурсия**: условия AND/OR обрабатываются рекурсивно с вложенными скобками
- **Типы значений**: int/float и str — оба используют плейсхолдер `$1`
- **Гибкость**: ORDER BY, LIMIT, OFFSET, JOIN, GROUP BY, HAVING поддерживаются опционально
