```lua
local BT = {}

BT.SUCCESS = "SUCCESS"
BT.FAILURE = "FAILURE"
BT.RUNNING = "RUNNING"

function BT.Action(fn)
    local node = {}
    function node:tick(ctx)
        local result = fn(ctx)
        if result == true or result == BT.SUCCESS then
            return BT.SUCCESS
        elseif result == false or result == BT.FAILURE then
            return BT.FAILURE
        else
            return BT.RUNNING
        end
    end
    function node:reset()
    end
    return node
end

function BT.Condition(predicate)
    local node = {}
    function node:tick(ctx)
        if predicate(ctx) then
            return BT.SUCCESS
        else
            return BT.FAILURE
        end
    end
    function node:reset()
    end
    return node
end

function BT.Sequence(children)
    local node = { children = children, index = 1 }
    function node:tick(ctx)
        for i = node.index, #node.children do
            local result = node.children[i]:tick(ctx)
            if result == BT.FAILURE then
                node.index = 1
                return BT.FAILURE
            elseif result == BT.RUNNING then
                node.index = i + 1
                return BT.RUNNING
            end
        end
        node.index = 1
        return BT.SUCCESS
    end
    function node:reset()
        node.index = 1
    end
    return node
end

function BT.Selector(children)
    local node = { children = children, index = 1 }
    function node:tick(ctx)
        for i = node.index, #node.children do
            local result = node.children[i]:tick(ctx)
            if result == BT.SUCCESS then
                node.index = 1
                return BT.SUCCESS
            elseif result == BT.RUNNING then
                node.index = i + 1
                return BT.RUNNING
            end
        end
        node.index = 1
        return BT.FAILURE
    end
    function node:reset()
        node.index = 1
    end
    return node
end

return BT
```
