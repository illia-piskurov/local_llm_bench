def run(program: str) -> list[str]:
    stack = []
    output = []
    variables = {}
    labels = {}
    pc = 0  # текущий индекс строки (позиция)
    call_stack = []  # стек вызовов подпрограмм
    
    lines = [line for line in program.splitlines() if line.strip() and not line.startswith('#')]
    
    # Собираем словарь меток и их позиций
    for idx, line in enumerate(lines, start=1):
        tokens = line.split()
        if not tokens:
            continue
        cmd = tokens[0].upper()
        
        # Определяем метку
        if cmd == 'LABEL':
            if len(tokens) != 2:
                raise ValueError(f"LABEL requires one argument at line {idx}")
            label_name = tokens[1]
            labels[label_name] = pc
            continue
        
        current_pc = pc
        
        # Обработка инструкций
        if cmd == 'PUSH':
            if len(tokens) != 2:
                raise ValueError(f"PUSH requires one argument at line {idx}")
            try:
                n = int(tokens[1])
            except ValueError:
                raise ValueError(f"Invalid number at line {idx}")
            stack.append(n)
        
        elif cmd == 'POP':
            if not stack:
                raise IndexError(f"Stack underflow at line {idx}")
            output.append(str(stack.pop()))
        
        elif cmd == 'ADD':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during ADD at line {idx}")
            b = stack.pop()
            a = stack.pop()
            stack.append(a + b)
        
        elif cmd == 'SUB':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during SUB at line {idx}")
            b = stack.pop()
            a = stack.pop()
            stack.append(a - b)
        
        elif cmd == 'MUL':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during MUL at line {idx}")
            b = stack.pop()
            a = stack.pop()
            stack.append(a * b)
        
        elif cmd == 'DIV':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during DIV at line {idx}")
            b = stack.pop()
            a = stack.pop()
            if b == 0:
                raise ZeroDivisionError(f"Division by zero at line {idx}")
            stack.append(a // b)
        
        elif cmd == 'DUP':
            if not stack:
                raise IndexError(f"Stack underflow during DUP at line {idx}")
            stack.append(stack[-1])
        
        elif cmd == 'SWAP':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during SWAP at line {idx}")
            a = stack.pop()
            b = stack.pop()
            stack.extend([a, b])
        
        elif cmd == 'PRINT':
            if not stack:
                raise IndexError(f"Stack empty for PRINT at line {idx}")
            output.append(str(stack[-1]))
        
        # Обработка подпрограммы
        if cmd == 'CALL':
            if len(tokens) != 2:
                raise ValueError(f"CALL requires one argument at line {idx}")
            target = tokens[1]
            if target not in labels:
                raise NameError(f"Subroutine {target} not found at line {idx}")
            call_stack.append(pc + 1)  # адрес возврата
            pc = labels[target]
            continue
        
        elif cmd == 'RET':
            if not call_stack:
                raise RuntimeError("RET without CALL at line {idx}")
            pc = call_stack.pop() - 1  # возвращаемся к инструкции после CALL
            continue
        
        # Операции сравнения
        elif cmd == 'EQ':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during EQ at line {idx}")
            b = stack.pop()
            a = stack.pop()
            stack.append(1 if a == b else 0)
        
        elif cmd == 'GT':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during GT at line {idx}")
            b = stack.pop()
            a = stack.pop()
            stack.append(1 if a > b else 0)
        
        elif cmd == 'LT':
            if len(stack) < 2:
                raise IndexError(f"Stack underflow during LT at line {idx}")
            b = stack.pop()
            a = stack.pop()
            stack.append(1 if a < b else 0)
        
        # Именованные переменные
        elif cmd == 'STORE':
            if len(tokens) != 2:
                raise ValueError(f"STORE requires one argument at line {idx}")
            var_name = tokens[1]
            if not stack:
                raise IndexError(f"Stack empty for STORE at line {idx}")
            variables[var_name] = stack.pop()
        
        elif cmd == 'LOAD':
            if len(tokens) != 2:
                raise ValueError(f"LOAD requires one argument at line {idx}")
            var_name = tokens[1]
            if var_name not in variables:
                raise NameError(f"Variable {var_name} not defined at line {idx}")
            stack.append(variables[var_name])
        
        # Обычные переходы
        if cmd in ('JMP', 'LABEL', 'JZ', 'JNZ'):
            if cmd == 'JMP':
                target = tokens[1]
                pc = labels.get(target, None)
                if pc is None:
                    raise NameError(f"Label {target} not found at line {idx}")
                continue
            elif cmd == 'LABEL':
                pass  # метка просто помечает позицию
            elif cmd == 'JZ':
                val = stack.pop() if stack else 0
                target = tokens[1]
                if val == 0:
                    pc = labels.get(target, None)
                    if pc is None:
                        raise NameError(f"Label {target} not found at line {idx}")
                continue
            elif cmd == 'JNZ':
                val = stack.pop() if stack else 0
                target = tokens[1]
                if val != 0:
                    pc = labels.get(target, None)
                    if pc is None:
                        raise NameError(f"Label {target} not found at line {idx}")
                continue
        
        pc += 1
    
    return output
