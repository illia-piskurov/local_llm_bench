```python
def run(program: str) -> list[str]:
    # Parse the program into instructions
    # Execute each instruction
    # Return list of printed strings
    
    # Stack-based language implementation
    stack = []
    
    for line in program.split('\n'):
        if not line.strip():
            continue  # Skip empty lines
        
        # Handle comments (lines starting with #)
        if line.startswith('#'):
            continue
        
        # Parse the instruction
        parts = line.split()
        
        if len(parts) < 2:
            raise ValueError(f"Instruction '{line}' has insufficient arguments")
        
        op = parts[0]
        args = parts[1:]
        
        try:
            if op == 'PUSH':
                stack.append(int(args[0]))
            elif op == 'POP':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.pop()
            elif op == 'ADD':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] + args[0])
            elif op == 'SUB':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] - args[0])
            elif op == 'MUL':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] * args[0])
            elif op == 'DIV':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1] // args[0])
            elif op == 'DUP':
                if len(stack) < 1:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.append(stack[-1])
            elif op == 'SWAP':
                if len(stack) < 2:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                stack.pop()
                stack.append(stack[-1])
            elif op == 'PRINT':
                if len(stack) < 1:
                    raise ValueError(f"Instruction '{line}' has insufficient arguments")
                print(stack[-1])
        except Exception as e:
            # Handle any unexpected errors
            pass
    
    return stack
```
