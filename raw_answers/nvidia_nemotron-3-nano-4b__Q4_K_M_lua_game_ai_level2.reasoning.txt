BT.Blackboard = {}
function BT.Blackboard.new()
    local bb = {
        data = {},
        watchers = {}, -- map key -> list of functions
        function get(self, key, default)
            return self.data[key] or default
        end,
        function set(self, key, value)
            if value == self.data[key] then return end
            self.data[key] = value
            local old = self.data[key]
            for _, cb in ipairs(self.watchers[key]) do cb(key, value, old) end
        end,
        function watch(self, key, callback)
            table.insert(self.watchers[key], callback)
        end
    }
    return bb
end
