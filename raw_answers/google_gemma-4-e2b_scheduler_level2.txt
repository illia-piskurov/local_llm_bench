def critical_path(tasks: dict[str, tuple[int, list[str]]]) -> int | None:
    """
    Calculates the length of the critical path (longest dependent sequence) 
    in a project network.

    Args:
        tasks: A dictionary where the key is the task name and the value is 
               a tuple (duration, dependencies_list).

    Returns:
        The length of the critical path, or None if a cycle is detected.
    """
    # 1. Build the graph and initialize necessary structures.
    graph = {}
    all_tasks = set()

    # Initialize graph structure and gather all unique tasks
    for task, (duration, dependencies) in tasks.items():
        all_tasks.add(task)
        graph[task] = []  # Initialize the task node in the adjacency list

        for dependency in dependencies:
            all_tasks.add(dependency)
            # Edge definition: Dependency -> Task (Dependency must precede Task)
            if dependency in graph:
                graph[dependency].append(task)
            # Handle case where a dependency might exist but is not explicitly defined as a task endpoint
            else:
                 graph[dependency] = [task]


    # State tracking for DFS (Topological Sort and Cycle Detection)
    visited = set()
    recursion_stack = set()
    topological_order = []

    def dfs(task: str) -> bool:
        """DFS helper for topological sort and cycle detection."""
        if task in recursion_stack:
            return False  # Cycle detected (back edge)
        if task in visited:
            return True   # Already processed

        recursion_stack.add(task)

        for neighbor in graph.get(task, []):
            if not dfs(neighbor):
                return False

        recursion_stack.remove(task)
        visited.add(task)
        topological_order.append(task)
        return True

    # 2. Perform DFS to get the topological order and check for cycles.
    for task in list(all_tasks):
        if task not in visited:
            if not dfs(task):
                return None  # Cycle found

    # If we reach here, the graph is a DAG (Directed Acyclic Graph).
    if not topological_order:
        return 0 # Empty graph

    # 3. Calculate the longest path duration using Dynamic Programming (DP).
    # longest_path_duration[T] stores the maximum duration of any path 
    # leading up to and including task T.
    longest_path_duration = {}

    def calculate_longest_path(task: str) -> int:
        """Calculates the longest path duration ending at 'task'."""
        if task in longest_path_duration:
            return longest_path_duration[task]

        duration, dependencies = tasks.get(task, (0, []))

        if not dependencies:
            # Base case: Start node
            longest_path_duration[task] = duration
            return duration

        # Recursive step: Find the maximum path length among all dependencies
        max_dependency_path = 0
        for dep in dependencies:
            # Note: Since we processed in topological order, dependency paths must be calculated
            max_dependency_path = max(max_dependency_path, calculate_longest_path(dep))
        
        # Total path length is current task duration + max incoming path
        current_path = duration + max_dependency_path
        longest_path_duration[task] = current_path
        return current_path

    # 4. Calculate the overall critical path length.
    max_duration = 0
    for task in topological_order:
        path_duration = calculate_longest_path(task)
        max_duration = max(max_duration, path_duration)

    return max_duration
