```html
<!DOCTYPE html>
<html lang="ru">
<head>
<meta charset="UTF-8">
<title>Змейка</title>
<style>
body{margin:0;overflow:hidden;background:#111;display:flex;justify-content:center;align-items:center;height:100vh;}
canvas{border:2px solid #fff;}
#score{position:absolute;top:10px;left:10px;font-size:24px;color:#fff;}
.overlay{
  position:absolute;top:0;left:0;width:100%;height:100%;
  background:rgba(0,0,0,0.8);display:flex;justify-content:center;align-items:center;
  color:#fff;font-size:48px;opacity:0;transition:opacity 0.5s ease-in-out;
}
@keyframes fadeIn{from{opacity:0}to{opacity:1}}
@keyframes fadeOut{from{opacity:1}to{opacity:0}}
</style>
</head>
<body>
<canvas id="game" width="400" height="400"></canvas>
<div id="score">Счёт: 0</div>
<div id="gameover" class="overlay">Game Over</div>

<script>
const canvas=document.getElementById('game');
const ctx=canvas.getContext('2d');
let snake=[];
let direction='RIGHT';
let speed=10;
let score=0;
let food={x:Math.floor(Math.random()*39)+1,y:Math.floor(Math.random()*39)};
let gameRunning=true;

let particles=[]; // {x,y,r,life}

function draw(){
  if(!gameRunning) return;
  ctx.clearRect(0,0,canvas.width,canvas.height);
  drawFood();
  drawSnake();
  drawParticles();
  requestAnimationFrame(draw);
}
function drawFood(){
  ctx.fillStyle='lime';
  ctx.fillRect(food.x*10,food.y*10,9,9);
}
function drawSnake(){
  for(let i=0;i<snake.length;i++){
    let x=snake[i].x*10;
    let y=snake[i].y*10;
    ctx.fillStyle='lime';
    ctx.fillRect(x,y,9,9);
    if(i===snake.length-1) ctx.strokeStyle='black'; else ctx.strokeStyle='none';
    ctx.strokeRect(x,y,9,9);
  }
}
function drawParticles(){
  for(let p of particles){
    let r=p.r+1;
    let life=p.life-r;
    if(life<=0) continue;
    ctx.fillStyle=`rgba(255,255,0,${life/200})`;
    ctx.beginPath();
    ctx.arc(p.x*10+p.r, p.y*10+p.r, r, 0, Math.PI*2);
    ctx.fill();
  }
}
function update(){
  let head={x:snake[0].x+getDirectionX(direction),y:snake[0].y+getDirectionY(direction)};
  if(head.x<0||head.x>39||head.y<0||head.y>39){
    gameRunning=false;
    document.getElementById('gameover').style.opacity='1';
    return;
  }
  for(let i=1;i<snake.length;i++){
    if(snake[i].x===head.x&&snake[i].y===head.y){
      gameRunning=false;
      document.getElementById('gameover').style.opacity='1';
      return;
    }
  }
  snake.unshift(head);
  if(head.x===food.x && head.y===food.y){
    score++;
    document.getElementById('score').textContent=`Счёт: ${score}`;
    // particles
    addParticle(food.x*10, food.y*10);
    // new food
    food={x:Math.floor(Math.random()*39)+1,y:Math.floor(Math.random()*39)};
    while(snake.some(seg=>seg.x===food.x&&seg.y===food.y)){
      food={x:Math.floor(Math.random()*39)+1,y:Math.floor(Math.random()*39)};
    }
  }else{
    snake.pop();
  }
}
function getDirectionX(dir){
  if(dir==='RIGHT') return 1;
  if(dir==='LEFT') return -1;
  return 0;
}
function getDirectionY(dir){
  if(dir==='DOWN') return 1;
  if(dir==='UP') return -1;
  return 0;
}
function addParticle(x,y){
  particles.push({x:x,y:y,r:0,life:200});
}
document.addEventListener('keydown',e=>{
  let key=e.key.toLowerCase();
  if(key==='w'||key==='ArrowUp'){
    direction='UP';
  }else if(key==='s'||key==='ArrowDown'){
    direction='DOWN';
  }else if(key==='a'||key==='ArrowLeft'){
    direction='LEFT';
  }else if(key==='d'||key==='ArrowRight'){
    direction='RIGHT';
  }
});
function restart(){
  snake=[{x:18,y:18}];
  score=0;
  document.getElementById('score').textContent=`Счёт: ${score}`;
  food={x:Math.floor(Math.random()*39)+1,y:Math.floor(Math.random()*39)};
  while(snake.some(seg=>seg.x===food.x&&seg.y===food.y)){
    food={x:Math.floor(Math.random()*39)+1,y:Math.floor(Math.random()*39)};
  }
  gameRunning=true;
  document.getElementById('gameover').style.opacity='0';
}
document.addEventListener('keypress',restart);
draw();
</script>
</body>
</html>
```
