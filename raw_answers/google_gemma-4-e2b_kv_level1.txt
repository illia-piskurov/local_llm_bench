import sys
from typing import Dict, Any, List, Tuple

class Transaction:
    """Represents an active transaction scope and its accumulated changes."""
    def __init__(self, parent: 'Transaction' = None):
        # Key -> Value (or marker for delete) changes pending in this transaction
        self.pending_changes: Dict[str, Any] = {}
        self.parent = parent

    def apply_change(self, key: str, value: Any):
        """Records a change within this transaction scope."""
        self.pending_changes[key] = value

    def commit(self):
        """
        Finalizes the changes made in this transaction.
        If a parent exists, these changes are merged into the parent's state.
        """
        if self.parent:
            for key, value in self.pending_changes.items():
                self.parent.apply_change(key, value)
        else:
            # This is the innermost transaction. Commit means applying changes to the global store.
            for key, value in self.pending_changes.items():
                # Apply directly to the global store (which is the implicit 'parent')
                self.parent.store[key] = value

    def rollback(self):
        """Discards all pending changes in this transaction."""
        self.pending_changes = {}

class KeyValueStore:
    def __init__(self):
        # Global storage for actual committed data
        self.store: Dict[str, Any] = {}
        # Stack to manage nested transactions. The base level is implicit (global store).
        self.transaction_stack: List[Transaction] = []

    def _get_current_transaction(self) -> Transaction | None:
        """Returns the current innermost transaction, or None if stack is empty."""
        if self.transaction_stack:
            return self.transaction_stack[-1]
        return None

    def set(self, key: str, value: Any):
        current_txn = self._get_current_transaction()

        if current_txn is None:
            # No transaction open, modify global store directly. (Immediate commit behavior)
            self.store[key] = value
        else:
            # Transaction open, record change in the transaction's delta.
            current_txn.apply_change(key, value)

    def get(self, key: str) -> Any:
        current_txn = self._get_current_transaction()

        if current_txn is None:
            # Read directly from global store
            return self.store.get(key, "NULL")
        else:
            # Read from the transaction's pending changes first.
            if key in current_txn.pending_changes:
                return current_txn.pending_changes[key]
            else:
                # If not in delta, check the committed state (from parent/global)
                if current_txn.parent:
                    return current_txn.parent.store.get(key, "NULL")
                else:
                    # Should ideally not happen if logic is clean, but handles initialization edge case
                    return "NULL"

    def delete(self, key: str):
        current_txn = self._get_current_transaction()

        if current_txn is None:
            # No transaction open, modify global store directly.
            if key in self.store:
                del self.store[key]
        else:
            # Transaction open, record the deletion in the transaction's delta.
            current_txn.apply_change(key, None) # Using None as a deletion marker

    def begin(self):
        """Starts a new transaction layer."""
        new_txn = Transaction(parent=None) # Root transactions are effectively the global store proxy
        self.transaction_stack.append(new_txn)

    def commit(self):
        current_txn = self._get_current_transaction()

        if current_txn is None:
            return # No transaction to commit

        current_txn.commit()
        self.transaction_stack.pop()


    def rollback(self):
        current_txn = self._get_current_transaction()

        if current_txn is None:
            return # No transaction to rollback

        current_txn.rollback()
        self.transaction_stack.pop()


def run(program: str) -> List[str]:
    """
    Processes a string of commands and returns the output for GET operations.
    """
    store = KeyValueStore()
    output: List[str] = []

    for line in program.strip().split('\n'):
        line = line.strip()
        if not line:
            continue

        parts = line.split()
        command = parts[0]

        if command == "SET":
            if len(parts) == 3:
                key, value = parts[1], parts[2]
                store.set(key, value)

        elif command == "GET":
            if len(parts) == 2:
                key = parts[1]
                result = store.get(key)
                output.append(str(result))

        elif command == "DELETE":
            if len(parts) == 2:
                key = parts[1]
                store.delete(key)

        elif command == "BEGIN":
            store.begin()

        elif command == "COMMIT":
            store.commit()
            # COMMIT produces no output

        elif command == "ROLLBACK":
            store.rollback()
            # ROLLBACK produces no output

    return output

if __name__ == '__main__':
    # Example usage demonstration (Not required for final submission, but useful for testing)
    
    print("--- Test Case 1: Basic Operations ---")
    program1 = """
SET a 10
GET a
DELETE a
GET a

"""
    results1 = run(program1)
    print("SET a 10, GET a, DELETE a, GET a\nOutput:", results1)
    # Expected: 10, NULL

    print("\n--- Test Case 2: Nested Transactions (Commit) ---")
    program2 = """
SET x 100
BEGIN
SET y 200  # Change 1 (pending)
GET y       # Should see 200

BEGIN
SET y 300  # Change 2 (pending)
GET y       # Should see 300

COMMIT      # Commit Change 2 (y=300)
GET y       # Should see 300

COMMIT      # Commit Change 1 (y=200)
GET y       # Should see 200

"""
    results2 = run(program2)
    print("SET x 100, BEGIN, SET y 200, GET y, BEGIN, SET y 300, GET y, COMMIT, GET y, COMMIT, GET y\nOutput:", results2)
    # Expected: ['NULL', '200', '300', '300', '200']

    print("\n--- Test Case 3: Nested Transactions (Rollback) ---")
    program3 = """
SET z 50
BEGIN
SET z 55  # Change 1 (pending)
GET z       # Should see 55

BEGIN
SET z 60  # Change 2 (pending)
GET z       # Should see 60

ROLLBACK    # Discard Change 2 (z=60)
GET z       # Should see 55

"""
    results3 = run(program3)
    print("SET z 50, BEGIN, SET z 55, GET z, BEGIN, SET z 60, GET z, ROLLBACK, GET z\nOutput:", results3)
    # Expected: ['55', '60', '55']

    print("\n--- Test Case 4: No Transaction ---")
    program4 = """
SET b 10
GET b
DELETE b
"""
    results4 = run(program4)
    print("SET b 10, GET b, DELETE b\nOutput:", results4)
    # Expected: [10, NULL]
